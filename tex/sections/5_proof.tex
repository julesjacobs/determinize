\section{Proof}
\label{sec:proof}

We now turn to the soundness of our determinization transformation. To motivate the proof strategy, first observe that at an $E$-mode distribution, the original program samples a value, whereas the transformed program computes the mean of the distribution. The resulting programs therefore have different output distributions. Thus, our proof relates both programs to a common symbolic execution, and the original and determiinzed programs are then obtained as two interpretations of the same symbolic execution. 

The symbolic semantics differs from the ordinary semantics only in its treatment of $\E$-mode draws. Rather than immediately sampling an $\E$-mode distribution or replacing it by its mean, symbolic execution records the draw using a fresh symbolic variable and continuous evaluating with that variable: wen symbolic execution encounters
$\draw D\E{\bar a}$
it does not sample from $D(\bar a)$. Instead, it introduces a fresh variable $u$, records the binding $u\sim D(\bar a)$ in a symbolic environment, and continues execution with $u$ in place of the sampled value. Dependencies between $\E$-mode draws are therefore preserved symbolically: the parameters of a later distribution may refer to variables introduced by earlier draws. By contrast, $\G$-mode draws are sampled immediately, exactly as in the ordinary semantics.

\textbf{Actual and expected interpretations}
\[
\begin{aligned}
  \mathsf{Act}_n(\sigma\mid\widehat e)
  &=
  \int
    \joint{
      \mathsf{realize}_\theta(\widehat e)
    }n\,
    \mathcal A_\sigma(d\theta),
  \\[6pt]
  \mathsf{Exp}_n(\sigma\mid\widehat e)
  &=
  \joint{
    \trans{
      \mathsf{realize}_{\bar\theta_\sigma}(\widehat e)
    }
  }n.
\end{aligned}
\]

Here $\mathcal A_\sigma$ is the joint distribution obtained by sampling
the declarations of $\sigma$ in order, and $\bar\theta_\sigma$ is the
sequential mean valuation. Thus $\mathsf{Act}_n(s)$ and
$\mathsf{Exp}_n(s)$ are measures over $\Trace\times\Real$, not over symbolic
configurations. The actual interpretation samples the recorded
$\E$-mode variables and then applies the trace-refined exact-depth semantics.
The expected interpretation substitutes their sequential means,
determinizes the resulting residual expression, and applies the same
trace-refined exact-depth semantics.

\begin{lemma}[Realization of initialization]
For every ordinary expression $e$,
\[
  \mathsf{realize}_{()}
  \bigl(\mathsf{lift}(e)\bigr)
  =
  e.
\]
\end{lemma}

\paragraph{Interpretation of initialized symbolic semantics.}
Recall that
\[
  \mathsf{init}(e)
  =
  (\emptyset\mid\mathsf{lift}(e)),
\]
where $\mathsf{lift}$ embeds an ordinary expression into the symbolic
language. The following theorems show that the two interpretations of the
symbolic semantics recover the source and transformed joint trace--output
measures.

\begin{theorem}[Actual interpretation of initialized symbolic execution]
\label{thm:actual-initialized-symbolic}
Suppose $\vdash e:\tyR\E$ and $e$ is closed. Then, for every
$n\in\Nat$,
\[
  \mathsf{Act}_n\bigl(\mathsf{init}(e)\bigr)
  =
  \joint e n.
\]
\end{theorem}


\begin{theorem}[Expected interpretation of initialized symbolic execution]
\label{thm:expected-initialized-symbolic}
Suppose $\vdash e:\tyR\E$ and $e$ is closed. Then, for every
$n\in\Nat$,
\[
  \mathsf{Exp}_n\bigl(\mathsf{init}(e)\bigr)
  =
  \joint{\trans e}n.
\]
\end{theorem}

\begin{theorem}[Agreement of interpretations]
\label{thm:agreement-of-interpretations}
Let
\[
  s^\tau=(\sigma\mid a:\tyR\E)
\]
be an integrable terminal symbolic configuration reached after fixing a
valid $\G$-trace $\tau$. Then $\mathsf{Act}_0(s^\tau)$ and
$\mathsf{Exp}_0(s^\tau)$ have finite first absolute moments, and
\[
  \int_{\Trace\times\Real}
    r\,\mathsf{Act}_0(s^\tau)(d\tau',dr)
  =
  \int_{\Trace\times\Real}
    r\,\mathsf{Exp}_0(s^\tau)(d\tau',dr).
\]
\end{theorem}


\begin{corollary}[Big-step interpretation of initialization]
Under the preceding assumptions,
\[
  \sum_{n=0}^{\infty}
    \mathsf{Act}_n\bigl(\mathsf{init}(e)\bigr)
  =
  J_e
\]
and
\[
  \sum_{n=0}^{\infty}
    \mathsf{Exp}_n\bigl(\mathsf{init}(e)\bigr)
  =
  J_{\trans e}.
\]
Consequently, for every measurable $B\subseteq\Real$,
\[
  \left(
    \sum_{n=0}^{\infty}
      \mathsf{Act}_n\bigl(\mathsf{init}(e)\bigr)
  \right)(\Trace\times B)
  =
  \law e(B)
\]
and
\[
  \left(
    \sum_{n=0}^{\infty}
      \mathsf{Exp}_n\bigl(\mathsf{init}(e)\bigr)
  \right)(\Trace\times B)
  =
  \law{\trans e}(B).
\]
\end{corollary}


\begin{theorem}[Generalized convex inequality]
\label{thm:generalized-convex-inequality}
Suppose $\vdash e:\tyR\E$, $e$ is closed and domain-safe, and
$\varphi:\Real\to\Real$ is nonnegative and convex. Then
\[
  \int_{\Real}
    \varphi(r)\,\law{\trans e}(dr)
  \le
  \int_{\Real}
    \varphi(r)\,\law e(dr),
\]
where the integrals are understood as nonnegative extended integrals.
\end{theorem}
