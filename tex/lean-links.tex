\newcommand{\DeclareLeanLink}[3]{%
  \expandafter\def\csname leanurl@#1\endcsname{\leanlinkbase/#2\leansourcesuffix\#L#3}}
% The link mark is the official Lean logo (the "LEAN" wordmark with the TM sign), used as
% provided by Lean FRO -- colours and TM sign included -- as its trademark
% policy requires (https://lean-lang.org/trademark-policy/). Source:
% https://lean-lang.org/logos/ (lean-logo-official-TM-transparent-2400x900.svg,
% kept next to the PDF); the PDF is the SVG rendered by rsvg-convert and
% trimmed to the drawing with pdfcrop. Placed like the Coq rooster in the
% Coq literature: a superscript-height mark right after the statement name.
% The wordmark is hairline art (its stroke is 3.6% of its height, 0.16pt at
% 1ex), so on its own it is barely visible on screen; a thin rounded frame
% around it, like the LEAN button in leanblueprint, makes it read as a mark
% without altering the logo itself.
\newcommand{\leanlogo}[1][1ex]{%
  \includegraphics[height=#1,alt={Lean}]{figures/logos/lean-logo}}
% Drawn with pgf primitives rather than a tikz node: \leanref is also used
% inside tikz nodes (figures/architecture.tex), and a nested node would
% inherit the outer node's text width, fill, and minimum size. The frame's
% padding and corner radius are in ex so the badge keeps its proportions in
% \small and \footnotesize text.
\newcommand{\leanbadge}{%
  \begingroup
  \setbox0=\hbox{\leanlogo}%
  \pgfpicture
    \pgfsetbaseline{0pt}%
    \pgfsetlinewidth{.3pt}%
    \pgfsetstrokecolor{black!50}%
    \pgfsetcornersarced{\pgfpoint{.25ex}{.25ex}}%
    \pgfpathrectanglecorners{\pgfpoint{-.3ex}{-.3ex}}{\pgfpoint{\wd0+.3ex}{\ht0+.3ex}}%
    \pgfusepath{stroke}%
    \pgftext[left,bottom]{\copy0}%
  \endpgfpicture
  \endgroup}
% \leanmark sets the mark after a name, with a gap a little wider than a
% thin space so the frame does not touch the last letter; \leanmark* is
% the bare mark for mentioning it in prose after a word space.
\newcommand{\leanmarksep}{\nobreak\hspace{.2em}}
\NewDocumentCommand{\leanmark}{s}{%
  \IfBooleanF{#1}{\leanmarksep}\raisebox{.3ex}{\leanbadge}}
\newcommand{\leanlink}[1]{%
  \leanmarksep\href{\csname leanurl@#1\endcsname}{\leanmark*}}
\DeclareRobustCommand{\leanref}[1]{%
  \ifleanlinks
    \ifcsname leanurl@#1\endcsname
      \ifmmode\smash{\text{\mbox{\leanlink{#1}}}}\else\smash{\mbox{\leanlink{#1}}}\fi
    \else
      \PackageError{lean-links}{Unknown Lean reference '#1'}{Add it to lean-links.tex.}%
    \fi
  \fi}
\pdfstringdefDisableCommands{\def\leanref#1{}}
