% \DeclareLeanLink{KEY}{TITLE}{NAMES}{FILE}{FIRST-LAST}: the paper's item TITLE is stated by the
% declarations NAMES, a comma-separated list of full Lean names, in the lines FIRST to LAST of
% FILE, a path from the root of the repository. The lines may also hold the declarations that
% NAMES rely on. A single line is given as {FIRST}.
\NewDocumentCommand{\DeclareLeanLink}{m m m m >{\SplitArgument{1}{-}}m}{%
  \leanlink@declare{#1}{#2}{#3}{#4}#5}
\newcommand{\leanlink@declare}[6]{%
  \ifcsname leanlink@file@#1\endcsname
    \PackageError{lean-links}{Lean reference '#1' declared twice}{}%
  \fi
  \@namedef{leanlink@title@#1}{#2}%
  \expandafter\edef\csname leanlink@names@#1\endcsname{\detokenize{#3}}%
  \expandafter\edef\csname leanlink@file@#1\endcsname{\detokenize{#4}}%
  \IfNoValueTF{#6}
    {\@namedef{leanlink@fragment@#1}{L#5}\@namedef{leanlink@lines@#1}{line~#5}}
    {\@namedef{leanlink@fragment@#1}{L#5-L#6}\@namedef{leanlink@lines@#1}{lines~#5--#6}}}

% The link mark is the official Lean logo (the "LEAN" wordmark with the TM sign), used as
% provided by Lean FRO -- colours and TM sign included -- as its trademark
% policy requires (https://lean-lang.org/trademark-policy/). Source:
% https://lean-lang.org/logos/ (lean-logo-official-TM-transparent-2400x900.svg,
% kept next to the PDF); the PDF is the SVG rendered by rsvg-convert and
% trimmed to the drawing with pdfcrop. Placed like the Coq rooster in the
% Coq literature: a superscript-height mark right after the statement name.
% The wordmark is hairline art (its stroke is 3.6% of its height, 0.16pt at
% 1ex), so on its own it is barely visible on screen; a thin rounded frame
% around it, like the LEAN button in leanblueprint, makes it read as a mark
% without altering the logo itself.
\newcommand{\leanlogo}[1][1ex]{%
  \includegraphics[height=#1,alt={Lean}]{figures/logos/lean-logo}}
% Drawn with pgf primitives rather than a tikz node: \leanref is also used
% inside tikz nodes (figures/architecture.tex), and a nested node would
% inherit the outer node's text width, fill, and minimum size. The frame's
% padding and corner radius are in ex so the badge keeps its proportions in
% \small and \footnotesize text.
\newcommand{\leanbadge}{%
  \begingroup
  \setbox0=\hbox{\leanlogo}%
  \pgfpicture
    \pgfsetbaseline{0pt}%
    \pgfsetlinewidth{.3pt}%
    \pgfsetstrokecolor{black!50}%
    \pgfsetcornersarced{\pgfpoint{.25ex}{.25ex}}%
    \pgfpathrectanglecorners{\pgfpoint{-.3ex}{-.3ex}}{\pgfpoint{\wd0+.3ex}{\ht0+.3ex}}%
    \pgfusepath{stroke}%
    \pgftext[left,bottom]{\copy0}%
  \endpgfpicture
  \endgroup}
% \leanmark sets the mark after a name, with a gap a little wider than a
% thin space so the frame does not touch the last letter; \leanmark* is
% the bare mark for mentioning it in prose after a word space.
\newcommand{\leanmarksep}{\nobreak\hspace{.2em}}
\NewDocumentCommand{\leanmark}{s}{%
  \IfBooleanF{#1}{\leanmarksep}\raisebox{.3ex}{\leanbadge}}
\newcommand{\leanlink}[1]{%
  \leanmarksep
  \ifleananonymous
    \leanlink@use{#1}%
    \hyperref[lean:#1]{\leanmark*}%
  \else
    \href{\leanrepository/blob/\leanrevision/\@nameuse{leanlink@file@#1}%
      \#\@nameuse{leanlink@fragment@#1}}{\leanmark*}%
  \fi}
\DeclareRobustCommand{\leanref}[1]{%
  \ifleanlinks
    \ifcsname leanlink@file@#1\endcsname
      \ifmmode\smash{\text{\mbox{\leanlink{#1}}}}\else\smash{\mbox{\leanlink{#1}}}\fi
    \else
      \PackageError{lean-links}{Unknown Lean reference '#1'}{Add it to lean-links.tex.}%
    \fi
  \fi}
\pdfstringdefDisableCommands{\def\leanref#1{}}

% The marks set so far, in the order of their first use, as \leanlink@entry{KEY}...
\newcommand*{\leanlink@used}{}
\newcommand{\leanlink@use}[1]{%
  \ifcsname leanlink@used@#1\endcsname\else
    \expandafter\gdef\csname leanlink@used@#1\endcsname{}%
    \g@addto@macro\leanlink@used{\leanlink@entry{#1}}%
  \fi}
% The list that the anonymous builds' marks link to: an entry for each mark set before it, in the
% order of first use, naming the declarations and their lines. A mark set after the list has no
% entry, and LaTeX warns of its undefined reference. Ragged right, as the long names leave too few
% spaces to justify the lines.
\newcounter{leanlink}
\newcommand{\printleanlinks}{%
  \ifx\leanlink@used\@empty\else
    \begin{list}{}{%
      \usecounter{leanlink}%
      \setlength{\leftmargin}{1em}%
      \setlength{\itemindent}{-\leftmargin}%
      \setlength{\labelwidth}{0pt}%
      \setlength{\labelsep}{0pt}%
      \raggedright}%
    \leanlink@used
    \end{list}%
  \fi}
% Lean names and paths in typewriter, broken after their dots and slashes.
\DeclareUrlCommand{\leancode}{\urlstyle{tt}}
% Each entry's link target is on its first line, which \phantomsection sets in the paragraph: the
% one that \item sets precedes the paragraph, and a page break can come between them.
\newcommand{\leanlink@entry}[1]{%
  \item\leavevmode\phantomsection\label{lean:#1}%
  \textbf{\@nameuse{leanlink@title@#1}.}
  \def\leanlink@separator{\def\leanlink@separator{, }}%
  \expanded{\noexpand\forcsvlist\noexpand\leanlink@name{\@nameuse{leanlink@names@#1}}}
  in \expanded{\noexpand\leancode{\@nameuse{leanlink@file@#1}}},
  \@nameuse{leanlink@lines@#1}.}
\newcommand{\leanlink@name}[1]{\leanlink@separator\leancode{#1}}
\makeatother
