\begin{lemma}[Type preservation]
  Let $e$ be a well-typed expression of type $\tau$, and assume that
  $\{e' \mid \emptyset \vdash e' : \tau'\}$ is measurable for every type $\tau'$.
  Then
  \[
    \sem{e : \tau}\bigl(\{\,e' \mid \emptyset \nvdash e' : \tau\,\}\bigr) = 0.
  \]
  That is, the small-step semantics assigns measure zero to the set of
  expressions that are not well-typed at type $\tau$.
\end{lemma}

\begin{lemma}[Step is a Markov kernel]
  The map $e \mapsto \sem{e : \tau}$ defines a Markov kernel $Expr_\tau \to \mathcal{D}(Expr_\tau)$.
\end{lemma}

\subsection{Big-step}

\begin{definition}[$n$-step semantics]
  The $n$-step semantics $\bigsemstep{e : \tau}{n}$ is defined by iterating the small-step kernel $n$ times:

  \[
    \bigsemstep{e : \tau}{n} =   
    \begin{cases}
      \delta_{e : \tau} &\text{ if } n = 0 \\
      \bigsemstep{e : \tau}{n-1} \monbind \lambda e'.\sem{e' : \tau} &\text{ if } n > 0.
    \end{cases}
  \]

\end{definition}

\begin{lemma}[Values are absorbing]\label{lem:values-absorbing}
  If $v : \tau \in Val_\tau$, then
  \[
    \sem{v : \tau} = \delta_{v : \tau}
    \qquad\text{and}\qquad
    \bigsemstep{v : \tau}{n} = \delta_{v : \tau}
  \]
  for every $n \in \mathbb{N}$.
\end{lemma}

\begin{proof}
  The first equality follows by inspection of values in
  \Cref{tab:small-step-semantics}. The second equality follows by induction on
  $n$, using the first equality and the left-unit law for Dirac measures in the
  successor case.
\end{proof}

\begin{lemma}[Monotone accumulation of mass]
  For all $n \in \mathbb{N}$:
  \begin{enumerate}
    \item $\bigsemstep{e : \tau}{n}(A) \leq \bigsemstep{e : \tau}{n+1}(A)$
  \end{enumerate}
  for every measurable $A \subseteq Val_\tau$. That is, probability mass on values accumulates monotonically.
\end{lemma}

\begin{definition}[Big-step semantics]
  For a closed, well-typed expression $e : \tau$ and every measurable
  $A \subseteq Val_\tau$, define
  \[
    \bigsem{e : \tau}(A)
    :=
    \lim_{n \to \infty}
    \bigsemstep{e : \tau}{n}(A).
  \]
\end{definition}

\subsection{Symbolic Expressions}
A symbolic state is:
\[
  s = (\sigma \mid \hat{e} : \tau) \in State_\tau.
\]

where $\sigma$ is the symbolic sample enviornment and $\hat{e} : \tau \in \widehat{Expr_\tau}$ is the well-typed symbolic residual expression. We elaborate on both below.

\subsubsection{Symbolic sample environment}
A symbolic sample environment records symbolic sampling operations in the
order in which they occur. Since the parameters of a sampling operation may
depend on previously introduced symbolic samples, we first define the affine
expressions permitted as parameters. 

For a finite set of symbolic variables $U=\{u_1,\ldots,u_n\}$, let
\[
  \mathsf{Aff}(U)
  =
  \left\{
    c_0+\sum_{\ell=1}^{n} c_\ell u_\ell
    \;\middle|\;
    c_0,\ldots,c_n\in\mathbb{R}
  \right\}.
\]
Thus, $\mathsf{Aff}(U)$ is the set of affine expressions over the variables
in $U$. In particular,
\[
  \mathsf{Aff}(\varnothing)=\mathbb{R}.
\]

\begin{definition}[Symbolic sample environment]
  The symbolic sample environment $\sigma$ is defined inductively:

  \textit{Base case.}
  The empty sequence $\emptyset$ is a symbolic sample environment.

  \textit{Inductive step.}
  Suppose
  \[
    \sigma
    =
    \bigl(
      u_1 \sim D_1(\vec a_1),\;
      \ldots,\;
      u_{i-1} \sim D_{i-1}(\vec a_{i-1})
    \bigr)
  \]
  is a symbolic sample environment. Let $D_i$ be a distribution with
  $r_i$ parameters
  \[
    \vec a_i=(a_{i,1},\ldots,a_{i,r_i}),
  \]
  where $r_i\in\{1,2\}$ for the distributions in our language, and where
  \[
    a_{i,j}\in\mathsf{Aff}(\{u_1,\ldots,u_{i-1}\})
    \qquad
    \text{for every }j\in\{1,\ldots,r_i\}.
  \]
  If $u_i\notin\{u_1,\ldots,u_{i-1}\}$, then
  \[
    \sigma,\;u_i\sim D_i(\vec a_i)
  \]
  is a symbolic sample environment.
\end{definition}

Thus, each distribution parameter may depend affinely on earlier symbolic
samples, but not on the current sample or any later sample. In particular,
the parameters of the first distribution are constants.

\begin{example}
  The symbolic sample environment
  \[
    \sigma
    =
    \bigl(
      u_1 \sim \uniform[\E](0,1),\;
      u_2 \sim \uniform[\E](u_1,u_1+1)
    \bigr).
  \]
  is valid.
  The parameters of the first distribution are constants, while the parameters
  $u_1$ and $u_1+1$ of the second distribution belong to
  $\mathsf{Aff}(\{u_1\})$. 
\end{example}

Suppose
\[
  \sigma =
  u_1 \sim D_1(\vec a_1),\ldots,
  u_n \sim D_n(\vec a_n),
  \qquad
  a_{i,j} \in \mathsf{Aff}(\{u_1,\ldots,u_{i-1}\}).
\]
Although $\sigma$ is an ordered context, it determines a distribution over
tuples of real numbers:
\[
  \sem{\sigma}_{\mathsf{env}} \in \mathcal{D}(\mathbb{R}^n).
\]

This distribution is defined sequentially by:
\[
\begin{aligned}
  \sem{\emptyset}_{\mathsf{env}}
  &= \delta_{()}, \\[0.5em]
  \sem{\sigma,\,u \sim D(\vec a)}_{\mathsf{env}}
  &=
  \sem{\sigma}_{\mathsf{env}}
  \monbind
  \lambda (r_1,\ldots,r_n).\,
  D\bigl(\vec a[r_1/u_1,\ldots,r_n/u_n]\bigr)
  \monbind
  \lambda r.\,
  \delta_{(r_1,\ldots,r_n,r)},
\end{aligned}
\]

The syntax for $\hat{e}$ is defined as follows:

    a ::= &\; c \mid u \mid a_1 + a_2 \mid c \cdot a & \text{affine expressions} \\

      | &\; \uniform[m](\hat e_1,\hat e_2) \mid \gaussian[m](\hat e_1,\hat e_2) & \text{distributions, } m \in \{\E,\G\} \\
	    | &\; \ifkw\; \hat e_1\; \thenkw\; \hat e_2\; \elsekw\; \hat e_3 & \text{conditionals} \\

    \ldots & \text{etc.}
	\end{align*}

Here, $\sym(a)$ is a symbolic value representing the affine expression $a$. The variable $u$ denotes a symbolic sample variable introduced by the symbolic sampling environment $\sigma$. The mode $m$ indicates whether the sample is expectation mode $\E$ or general mode $\G$. All other constructs follow the original syntax.

\paragraph{Type System}

In a given symbolic state, only the symbolic residual expression is typed. The type system has a single judgment form:
\[
  \Gamma \vdash \hat e : \tau.
\]

The typing rules from the ordinary expresisons lift to symbolic residual expressions. 
\begin{mathpar}
  \inferrule[\textsc{Var}]
    {x:\tau \in \Gamma}
    {\Gamma \vdash x : \tau}
  \and
  \inferrule[\textsc{Sym}]
    {a\ \text{is affine}}
    {\Gamma \vdash \sym(a) : \Float{m}}
  \and
  \inferrule[\textsc{Let}]
    {\Gamma \vdash \hat e_1 : \tau_1
      \\ \Gamma,x:\tau_1 \vdash \hat e_2 : \tau_2}
    {\Gamma \vdash
      \letkw\;x=\hat e_1\;\inkw\;\hat e_2 : \tau_2}
  \and
  \inferrule[\textsc{Pair}]
    {\Gamma \vdash \hat e_1 : \tau_1
      \\ \Gamma \vdash \hat e_2 : \tau_2}
    {\Gamma \vdash
      \langle \hat e_1,\hat e_2\rangle : \tau_1 \times \tau_2}
  \and
  \inferrule[\textsc{Add}]
    {\Gamma \vdash \hat e_1 : \Float{m_1}
      \\ \Gamma \vdash \hat e_2 : \Float{m_2}
      \\ m_1 \preceq m
      \\ m_2 \preceq m}
    {\Gamma \vdash \hat e_1 + \hat e_2 : \Float{m}}
  \and
  \inferrule[\textsc{Uniform}]
    {\Gamma \vdash \hat e_1 : \Float{m_1}
      \\ \Gamma \vdash \hat e_2 : \Float{m_2}
      \\ m_1 \preceq m
      \\ m_2 \preceq m}
    {\Gamma \vdash \uniform[m](\hat e_1,\hat e_2) : \Float{m}}
  \and
  \inferrule[\textsc{Less}]
    {\Gamma \vdash \hat e_1 : \Float{m_1}
      \\ \Gamma \vdash \hat e_2 : \Float{m_2}
      \\ m_1 \preceq \G
      \\ m_2 \preceq \G}
    {\Gamma \vdash \hat e_1 < \hat e_2 : \Bool}
\end{mathpar}

\paragraph{Lifting an ordinary expression to a symbolic residual expression}
We can define a symbolic lifting:
\[
\mathsf{lift} :
Expr_\tau \longrightarrow \widehat{Expr}_\tau
\]
which maps an ordinary expression $e : \tau$ to a symbolic residual expression $\hat e : \tau$ of the same type. It acts structurally on expressions, replacing each floating point constant $c$ with the symbolic constant $\sym(c)$.
\begin{align*}
\mathsf{lift}(x) &= x,\\
\mathsf{lift}(c:\Float{m}) &= \sym(c):\Float{m},\\
\mathsf{lift}(\mathsf{true}) &= \mathsf{true},\\
\mathsf{lift}(\mathsf{false}) &= \mathsf{false},\\
\mathsf{lift}(e_1+e_2)
  &= \mathsf{lift}(e_1)+\mathsf{lift}(e_2),\\
\mathsf{lift}(\letkw\ x=e_1\ \inkw\ e_2)
  &=\letkw\ x=\mathsf{lift}(e_1)\
    \inkw\ \mathsf{lift}(e_2),\\
\mathsf{lift}(\ifkw\ e_1\ \thenkw\ e_2\ \elsekw\ e_3)
  &=\ifkw\ \mathsf{lift}(e_1)\
    \thenkw\ \mathsf{lift}(e_2)\
    \elsekw\ \mathsf{lift}(e_3),\\
\mathsf{lift}\bigl(
  \uniform(e_1,e_2):\Float{m}
\bigr)
  &=
  \uniform[m]\bigl(
    \mathsf{lift}(e_1),
    \mathsf{lift}(e_2)
  \bigr):\Float{m} \\
  \ldots etc.
\end{align*}

\subsubsection{Symbolic semantics}
We start with the following notation for the one-step symbolic semantics. The small-step symbolic semantics is given in~\Cref{tab:symbolic-small-step-semantics}.

\[
  \symsem{s : \tau} \in \mathcal{D}(State_\tau)
\]

Now, define the $n$-step symbolic semantics as follows:

\begin{align*}
  \symbigsemstep{s : \tau}{n} =
  \begin{cases}
    \delta_{s : \tau} &\text{ if } n = 0 \\
    \symbigsemstep{s : \tau}{n-1} \monbind \lambda s. \symsem{s : \tau} &\text{ if } n > 0
  \end{cases}
\end{align*}

where $\symbigsemstep{s : \tau}{n} \in \mathcal{D}(State_\tau)$.

\begin{example}
  \[
  \begin{aligned}
    &(\emptyset \mid \uniform[\E](0,1) : \Float{\E} + \sym(1)) \\
    &\longrightarrow
	    (u_1 \sim \uniform[\E](0,1)
	      \mid \sym(u_1) + \sym(1)) \\
	    &\longrightarrow
	    (u_1 \sim \uniform[\E](0,1)
	      \mid \sym(u_1 + 1)).
  \end{aligned}
  \]
\end{example}

To evaluate an ordinary expression symbolically, we first embed it into the space of symbolic states. We capture this initial state with the following:

\begin{definition}[Initialization]
  For every type $\tau$, the initialization function
  \[
    \mathsf{init}_\tau:
    Expr_\tau \to State_\tau
  \]
  is defined by
  \[
    \mathsf{init}_\tau(e:\tau)
    =
    (\emptyset\mid\mathsf{lift}(e:\tau):\tau).
  \]
\end{definition}

The function $\mathsf{init}_\tau$ is applied only at the beginning of symbolic
evaluation; subsequent symbolic states are produced by the symbolic small-step
semantics.

\subsubsection{Actual interpretation of symbolic semantics}

The actual interpretation turns a symbolic state back into a distribution over
ordinary expressions by sampling the symbolic environment and substituting the
sampled values for the symbolic variables in the residual expression.  


Recall $\sym(a)$ embeds an affine expression $a$ into the
language of symbolic residual expressions. We equip this auxiliary language
with a big-step semantics.

Let $U$ be a finite set of symbolic sample variables. A valuation for $U$
is a function
\[
  \rho : U \to \mathbb{R}
\]
that assigns a concrete real value to each variable in $U$. For
$a\in\mathsf{Aff}(U)$, the judgment
\[
  \rho \vdash a \Downarrow r
\]
means that $a$ evaluates to $r\in\mathbb{R}$ under $\rho$. It is defined
by the following rules:
\begin{mathpar}
  \inferrule[\textsc{A-Const}]
    {\,}
    {\rho \vdash c \Downarrow c}
  \and
  \inferrule[\textsc{A-Var}]
    {u\in U}
    {\rho \vdash u \Downarrow \rho(u)}
  \and
  \inferrule[\textsc{A-Add}]
    {\rho \vdash a_1 \Downarrow r_1 \\
     \rho \vdash a_2 \Downarrow r_2}
    {\rho \vdash a_1+a_2 \Downarrow r_1+r_2}
  \and
  \inferrule[\textsc{A-Scale}]
    {\rho \vdash a \Downarrow r}
    {\rho \vdash c\cdot a \Downarrow c\cdot r}
\end{mathpar}

We can therefore use this semantics to realize symbolic values. 

\begin{definition}[Realization]
Given a
valuation $\rho$, the realization function
\[
  \mathsf{realize}_\rho :
  \widehat{Expr_\tau}\to Expr_\tau
\]
is defined recursively on symbolic residual expressions:
  \begin{align*}
    \mathsf{realize}_\rho(x)
    &=
    x,\\
    \mathsf{realize}_\rho(\text{true})
    &=
    \text{true},\\
    \mathsf{realize}_\rho(\text{false})
    &=
    \text{false},\\
    \mathsf{realize}_\rho(\sym(a))
    &=
    r
    \qquad
    \text{whenever } \rho \vdash a \Downarrow r,\\
    \mathsf{realize}_\rho(\letkw\ x=\hat e_1\ \inkw\ \hat e_2)
    &=
    \letkw\ x=\mathsf{realize}_\rho(\hat e_1)
    \ \inkw\ \mathsf{realize}_\rho(\hat e_2),\\
    \mathsf{realize}_\rho(
      \ifkw\ \hat e_1\ \thenkw\ \hat e_2\ \elsekw\ \hat e_3)
    &=
    \ifkw\ \mathsf{realize}_\rho(\hat e_1)
    \ \thenkw\ \mathsf{realize}_\rho(\hat e_2)
    \ \elsekw\ \mathsf{realize}_\rho(\hat e_3),\\
    \mathsf{realize}_\rho(\hat e_1 + \hat e_2)
    &=
    \mathsf{realize}_\rho(\hat e_1)
    +
    \mathsf{realize}_\rho(\hat e_2),\\
    \mathsf{realize}_\rho
      \bigl(\uniform[m](\hat e_1,\hat e_2)\bigr)
    &=
    \uniform\bigl(
      \mathsf{realize}_\rho(\hat e_1),
      \mathsf{realize}_\rho(\hat e_2)
    \bigr) \\
    \ldots
  \end{align*}
\end{definition}

Strictly speaking, $\mathsf{realize}_{\rho}$ acts on fully type-annotated symbolic residual expressions and preserves their type annotations, but we omit the types here for less notational clutter.

\begin{example}
  \[
    \mathsf{realize}_{\rho}\bigl(\sym(2+1)\bigr)
    =
    3.
  \]
\end{example}

\begin{example}
  Let $\rho=[u_1\mapsto 2,\;u_2\mapsto 4]$. Then
  \[
    \begin{aligned}
      \mathsf{realize}_{(2,4)}
      \bigl(\sym(3u_1+u_2+1)\bigr)
      &=
      3 \cdot 2 + 4 + 1 = 11.
    \end{aligned}
  \]
\end{example}

\begin{lemma}\label{lem:realize-lift}
  \[
    \mathsf{realize}_{()}(\mathsf{lift}(e : \tau)) = e : \tau
  \]
\end{lemma}

\begin{proof}
  By induction on $e : \tau$. 

  \paragraph{Case $e=x:\tau$.}
  \begin{align*}
    \mathsf{realize}_{()}
    \bigl(\mathsf{lift}(x:\tau)\bigr)
    &=
    \mathsf{realize}_{()}(x:\tau)
    \tag{by the definition of $\mathsf{lift}$}
    \\
    &=
    x:\tau
    \tag{by the definition of $\mathsf{realize}_{()}$}
  \end{align*}

  \paragraph{Case $e=c:\Float m$.}
  \begin{align*}
    \mathsf{realize}_{()}
    \bigl(\mathsf{lift}(c:\Float m)\bigr)
    &=
    \mathsf{realize}_{()}
    \bigl(\sym(c):\Float m\bigr)
    \tag{by the definition of $\mathsf{lift}$}
    \\
    &=
    c:\Float m
    \tag{since $()\vdash c\Downarrow c$}
  \end{align*}

  \paragraph{Case $e=e_1+e_2:\Float m$.}
  \begin{align*}
    \mathsf{realize}_{()}
    \bigl(\mathsf{lift}(e_1+e_2)\bigr)
    &=
    \mathsf{realize}_{()}
    \bigl(
      \mathsf{lift}(e_1)+\mathsf{lift}(e_2)
    \bigr)
    \tag{by the definition of $\mathsf{lift}$}
    \\
    &=
    \mathsf{realize}_{()}
    \bigl(\mathsf{lift}(e_1)\bigr)
    +
    \mathsf{realize}_{()}
    \bigl(\mathsf{lift}(e_2)\bigr)
    \tag{by the definition of $\mathsf{realize}_{()}$}
    \\
    &=
    e_1+e_2
    \tag{by the I.H.}
  \end{align*}

  \paragraph{All other cases analogous.}
\end{proof}


\begin{definition}[Single-state actual interpretation]
  The single-state actual interpretation is the function:
  \[
    \mathsf{act}_\tau : State_\tau \to \mathcal{D}(Expr_\tau)
  \]
  defined by
  \[
  \begin{aligned}
    \mathsf{act}_\tau\big(\ (\sigma \mid \hat e)\ \big)
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \delta_{\mathsf{realize}_\rho(\hat e) : \tau}.
  \end{aligned}
  \]
  where a tuple $(r_1,\ldots,r_n)$ sampled from
  $\sem{\sigma}_{\mathsf{env}}$ is identified with the valuation
  $\rho=[u_1\mapsto r_1,\ldots,u_n\mapsto r_n]$.
\end{definition}


\begin{lemma}[Initial actual interpretation]
  Let $e:\tau$ be closed and well typed. Then
  \[
    \mathsf{act}_\tau
    (\mathsf{init}_\tau(e : \tau))
    =
    \delta_{e:\tau}.
  \]
\end{lemma}

\begin{proof}
  \begin{align}
    \mathsf{act}_\tau
    \bigl(\mathsf{init}_\tau(e:\tau)\bigr)
    &=
    \mathsf{act}_\tau
    \bigl(\emptyset\mid\mathsf{lift}(e:\tau):\tau\bigr)
    \tag{by the definition of $\mathsf{init}_\tau$}
    \\
    &=
    \sem{\emptyset}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \delta_{\mathsf{realize}_\rho(\mathsf{lift}(e:\tau)):\tau}
    \tag{by the definition of $\mathsf{act}_\tau$}
    \\
    &=
    \delta_{()}
    \monbind
    \lambda\rho.\,
    \delta_{\mathsf{realize}_\rho(\mathsf{lift}(e:\tau)):\tau}
    \tag{by the definition of $\sem{\emptyset}_{\mathsf{env}}$}
    \\
    &=
    \delta_{\mathsf{realize}_{()}(\mathsf{lift}(e:\tau)):\tau}
    \tag{by monad left identity}
    \\
    &=
    \delta_{e:\tau}
    \tag{by~\Cref{lem:realize-lift}}
  \end{align}
\end{proof}

The following lemma states that taking one symbolic step and then actualizing the resulting symbolic state yields the same distribution as first actualizing the original state and then taking one ordinary step. 

\begin{lemma}[Actual interpretation of symbolic small-step semantics]\label{lem:symbolic-step}
  Let $s=(\sigma \mid \hat e : \tau) \in State_\tau$, then, for all $\sigma$,
  \begin{equation}
    \symsem{s : \tau}
    \monbind
    \mathsf{act}_\tau
    =
    \mathsf{act}_\tau(s)
    \monbind
    \lambda e'.\,\sem{e' : \tau}
  \end{equation}
\end{lemma}

\begin{proof}
  By structural induction on the symbolic residual expression $\hat e$,
  with the statement quantified over all $\sigma$. We show the conditional
  case; the remaining cases are analogous.

  Suppose
  \[
    \hat e
    =
    \ifkw\ \hat e_1\ \thenkw\ \hat e_2\ \elsekw\ \hat e_3,
  \]
  where
  \[
    \vdash \hat e_1 : \Bool
    \qquad\text{and}\qquad
    \vdash \hat e_2,\hat e_3 : \tau.
  \]

  We proceed by cases on $\hat e_1$.

  \paragraph{Case $\hat e_1 = \mathsf{true}$.}
  For the left-hand side,
  \begin{align}
    &\symsem{
      (\sigma \mid
      \ifkw\ \mathsf{true}\ \thenkw\ \hat e_2\
      \elsekw\ \hat e_3 : \tau)
    }
    \monbind
    \lambda(\sigma' \mid \hat e').\,
    \mathsf{act}_\tau(\sigma' \mid \hat e')
    \notag \\
    &\qquad=
    \delta_{(\sigma \mid \hat e_2 : \tau)}
    \monbind
    \lambda(\sigma' \mid \hat e').\,
    \mathsf{act}_\tau(\sigma' \mid \hat e')
    \tag{\text{by symbolic $\mathsf{if}$ rule}} \\
    &\qquad=
    \mathsf{act}_\tau(\sigma \mid \hat e_2 : \tau)
    \tag{\text{by monad left identity}} \\
    &\qquad=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \delta_{\mathsf{realize}_\rho(\hat e_2) : \tau}.
    \tag{\text{by definition of $\mathsf{act}_\tau$}}
  \end{align}
  For the right-hand side,
  \begin{align}
    &\mathsf{act}_\tau
    \bigl(
      \sigma
      \mid
      \ifkw\ \mathsf{true}\ \thenkw\ \hat e_2\
      \elsekw\ \hat e_3 : \tau
    \bigr)
    \monbind
    \lambda e'.\,\sem{e' : \tau}
    \notag \\
    &\qquad=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \sem{
      \mathsf{realize}_\rho
      \bigl(
        \ifkw\ \mathsf{true}\ \thenkw\ \hat e_2\
        \elsekw\ \hat e_3
      \bigr)
      : \tau
    }
    \tag{\text{by definition of $\mathsf{act}_\tau$}} \\
    &\qquad=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \sem{
      \ifkw\ \mathsf{true}\
      \thenkw\ \mathsf{realize}_\rho(\hat e_2)\
      \elsekw\ \mathsf{realize}_\rho(\hat e_3)
      : \tau
    }
    \tag{\text{by $\mathsf{realize}_\rho$ on $\mathsf{if}$}} \\
    &\qquad=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \delta_{\mathsf{realize}_\rho(\hat e_2) : \tau}.
    \tag{\text{by ordinary $\mathsf{if}$ rule}}
  \end{align}
  The two sides agree.

  \paragraph{Case $\hat e_1 = \mathsf{false}$.} Analogous to above. 

  \paragraph{Case $\hat e_1$ is not a Boolean value.}
  Abbreviate $\mathsf{If}(\hat g) := \ifkw\ \hat g\ \thenkw\ \hat e_2\
  \elsekw\ \hat e_3$. The induction hypothesis for the subexpression $\hat e_1$ is:
  \begin{equation}
    \label{eq:if-ih}
    \symsem{(\sigma \mid \hat e_1 : \Bool)}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
    \mathsf{act}_{\Bool}(\sigma' \mid \hat g)
    =
    \mathsf{act}_{\Bool}(\sigma \mid \hat e_1)
    \monbind
    \lambda b.\,\sem{b : \Bool}.
  \end{equation}
  
  We start from the left-hand side:

  \begin{align}
    &\symsem{(\sigma \mid \mathsf{If}(\hat e_1) : \tau)}
    \monbind
    \lambda(\sigma' \mid \hat e').\,
    \mathsf{act}_\tau(\sigma' \mid \hat e')
    \notag \\
    &\quad=
    \left(
      \symsem{(\sigma \mid \hat e_1 : \Bool)}
      \monbind
      \lambda(\sigma' \mid \hat g).\,
      \delta_{(\sigma' \mid \mathsf{If}(\hat g) : \tau)}
    \right)
    \monbind
    \lambda(\sigma' \mid \hat e').\,
    \mathsf{act}_\tau(\sigma' \mid \hat e')
    \tag{\text{by symbolic $\mathsf{if}$ rule}} \\
    &\quad=
    \symsem{(\sigma \mid \hat e_1 : \Bool)}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
    \left(
      \delta_{(\sigma' \mid \mathsf{If}(\hat g) : \tau)}
      \monbind
      \lambda(\sigma'' \mid \hat e'').\,
      \mathsf{act}_\tau(\sigma'' \mid \hat e'')
    \right)
    \tag{\text{by associativity}} \\
    &\quad=
    \symsem{(\sigma \mid \hat e_1 : \Bool)}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
    \mathsf{act}_\tau(\sigma' \mid \mathsf{If}(\hat g) : \tau)
    \tag{\text{by left identity}} \\
    &\quad=
    \symsem{(\sigma \mid \hat e_1 : \Bool)}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
    \left(
      \sem{\sigma'}_{\mathsf{env}}
      \monbind
      \lambda\rho.\,
      \delta_{\mathsf{realize}_\rho(\mathsf{If}(\hat g)) : \tau}
    \right)
    \tag{\text{by definition of $\mathsf{act}_\tau$}} \\
    &\quad=
    \symsem{(\sigma \mid \hat e_1 : \Bool)}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
    \left(
      \sem{\sigma'}_{\mathsf{env}}
      \monbind
      \lambda\rho.\,
      \delta_{\mathsf{If}(\mathsf{realize}_\rho(\hat g)) : \tau}
    \right)
    \tag{\text{by $\mathsf{realize}_\rho$ on $\mathsf{if}$}} \\
    &\quad=
    \symsem{(\sigma \mid \hat e_1 : \Bool)}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
    \left(
      \sem{\sigma'}_{\mathsf{env}}
      \monbind
      \lambda\rho.\,
      \left(
        \delta_{\mathsf{realize}_\rho(\hat g) : \Bool}
        \monbind
        \lambda b.\,\delta_{\mathsf{If}(b) : \tau}
      \right)
    \right)
    \tag{\text{by left identity}} \\
    &\quad=
    \symsem{(\sigma \mid \hat e_1 : \Bool)}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
    \left(
      \underbrace{
        \left(
          \sem{\sigma'}_{\mathsf{env}}
          \monbind
          \lambda\rho.\,
          \delta_{\mathsf{realize}_\rho(\hat g) : \Bool}
        \right)
      }_{=\,\mathsf{act}_{\Bool}(\sigma' \mid \hat g)}
      \monbind
      \lambda b.\,\delta_{\mathsf{If}(b) : \tau}
    \right)
    \tag{\text{by associativity}} \\
    &\quad=
    \symsem{(\sigma \mid \hat e_1 : \Bool)}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
    \left(
      \mathsf{act}_{\Bool}(\sigma' \mid \hat g)
      \monbind
      \lambda b.\,\delta_{\mathsf{If}(b) : \tau}
    \right)
    \tag{\text{by definition of $\mathsf{act}_{\Bool}$}} \\
    &\quad=
    \left(
      \symsem{(\sigma \mid \hat e_1 : \Bool)}
      \monbind
      \lambda(\sigma' \mid \hat g).\,
      \mathsf{act}_{\Bool}(\sigma' \mid \hat g)
    \right)
    \monbind
    \lambda b.\,\delta_{\mathsf{If}(b) : \tau}
    \tag{\text{by associativity}}
  \end{align}
  The head of the expression is now exactly the left-hand side of the induction
  hypothesis~\eqref{eq:if-ih}. Applying it:
  \begin{align}
    &\quad= \left(
      \mathsf{act}_{\Bool}(\sigma \mid \hat e_1)
      \monbind
      \lambda b.\,\sem{b : \Bool}
    \right)
    \monbind
    \lambda b.\,\delta_{\mathsf{If}(b) : \tau}
    \tag{\text{by \eqref{eq:if-ih}}} \\
    &\quad=
    \mathsf{act}_{\Bool}(\sigma \mid \hat e_1)
    \monbind
    \lambda b.\,
    \left(
      \sem{b : \Bool}
      \monbind
      \lambda b'.\,\delta_{\mathsf{If}(b') : \tau}
    \right)
    \tag{\text{by associativity}} \\
    &\quad=
    \left(
      \sem{\sigma}_{\mathsf{env}}
      \monbind
      \lambda\rho.\,
      \delta_{\mathsf{realize}_\rho(\hat e_1) : \Bool}
    \right)
    \monbind
    \lambda b.\,
    \left(
      \sem{b : \Bool}
      \monbind
      \lambda b'.\,\delta_{\mathsf{If}(b') : \tau}
    \right)
    \tag{\text{by definition of $\mathsf{act}_{\Bool}$}} \\
    &\quad=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \left(
      \delta_{\mathsf{realize}_\rho(\hat e_1) : \Bool}
      \monbind
      \lambda b.\,
      \left(
        \sem{b : \Bool}
        \monbind
        \lambda b'.\,\delta_{\mathsf{If}(b') : \tau}
      \right)
    \right)
    \tag{\text{by associativity}} \\
    &\quad=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \left(
      \sem{\mathsf{realize}_\rho(\hat e_1) : \Bool}
      \monbind
      \lambda b'.\,\delta_{\mathsf{If}(b') : \tau}
    \right)
    \tag{\text{by left identity}} \\
    &\quad=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \sem{
      \ifkw\ \mathsf{realize}_\rho(\hat e_1)\
      \thenkw\ \mathsf{realize}_\rho(\hat e_2)\
      \elsekw\ \mathsf{realize}_\rho(\hat e_3)
      : \tau
    }
    \tag{\text{by ordinary $\mathsf{if}$ rule}} \\
    &\quad=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \sem{\mathsf{realize}_\rho(\mathsf{If}(\hat e_1)) : \tau}
    \tag{\text{by $\mathsf{realize}_\rho$ on $\mathsf{if}$}} \\
    &\quad=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \left(
      \delta_{\mathsf{realize}_\rho(\mathsf{If}(\hat e_1)) : \tau}
      \monbind
      \lambda e'.\,\sem{e' : \tau}
    \right)
    \tag{\text{by left identity}} \\
    &\quad=
    \left(
      \sem{\sigma}_{\mathsf{env}}
      \monbind
      \lambda\rho.\,
      \delta_{\mathsf{realize}_\rho(\mathsf{If}(\hat e_1)) : \tau}
    \right)
    \monbind
    \lambda e'.\,\sem{e' : \tau}
    \tag{\text{by associativity}} \\
    &\quad=
    \mathsf{act}_\tau(\sigma \mid \mathsf{If}(\hat e_1) : \tau)
    \monbind
    \lambda e'.\,\sem{e' : \tau}.
    \tag{\text{by definition of $\mathsf{act}_\tau$}}
  \end{align}
  This is the right-hand side, completing the conditional case.
\end{proof}



\begin{theorem}[Actual interpretation of symbolic $n$-step semantics]
  \label{thm:actual-n-step}
  If $s = (\sigma \mid \hat e : \tau) \in State_\tau$, then for
  every $n \in \mathbb{N}$,
  \[
    \symbigsemstep{s : \tau}{n}
    \monbind
    \mathsf{act}_\tau
    =
    \mathsf{act}_\tau(s)
    \monbind
    \lambda e.\,\bigsemstep{e : \tau}{n}.
  \]
\end{theorem}

\begin{proof}
  The proof is by induction on $n$, with the statement quantified over all
  well-typed symbolic states.

  For $n=0$, let $s \in State_\tau$. Then
  \begin{align}
    \symbigsemstep{s : \tau}{0}
    \monbind
    \mathsf{act}_\tau
    &=
    \delta_{s : \tau}
    \monbind
    \mathsf{act}_\tau
    \tag{\text{by definition of n-step symbolic semantics}} \\
    &=
    \mathsf{act}_\tau(s)
    \tag{\text{by monad left identity}} \\
    &=
    \mathsf{act}_\tau(s)
    \monbind
    \lambda e.\,\delta_{e : \tau}
    \tag{\text{by monad right identity}} \\
    &=
    \mathsf{act}_\tau(s)
    \monbind
    \lambda e.\,\bigsemstep{e : \tau}{0}.
    \tag{\text{by definition of n-step semantics}}
  \end{align}

  For the inductive step, assume the statement holds for $n$ for every
  well-typed symbolic state, and let
  $s = (\sigma \mid \hat e : \tau) \in State_\tau$. We use
  \Cref{lem:symbolic-step}:

  \begin{equation}
    \symsem{s : \tau}
    \monbind
    \mathsf{act}_\tau
    =
    \mathsf{act}_\tau(s)
    \monbind
    \lambda e'.\,\sem{e' : \tau}
  \end{equation}
  for every well-typed symbolic state $s \in State_\tau$. Then
  \begin{align}
    &\symbigsemstep{s : \tau}{n+1}
    \monbind
    \mathsf{act}_\tau \notag \\
    &=
    \left(
      \symbigsemstep{s : \tau}{n}
      \monbind
      \lambda s'.\,\symsem{s' : \tau}
    \right)
    \monbind
    \mathsf{act}_\tau
    \tag{\text{by definition of symbolic n-step}} \\
    &=
    \symbigsemstep{s : \tau}{n}
    \monbind
    \lambda s'.\,
      \left(\symsem{s' : \tau}
      \monbind \mathsf{act}_\tau\right)
    \tag{\text{by monad associativity}} \\
    &=
    \symbigsemstep{s : \tau}{n}
    \monbind
    \lambda s'.\,
      \left(\mathsf{act}_\tau(s')
      \monbind \lambda e'.\,\sem{e' : \tau}\right)
    \tag{\text{by \Cref{lem:symbolic-step}}} \\
    &=
    \left(
      \symbigsemstep{s : \tau}{n}
      \monbind
      \lambda s'.\,\mathsf{act}_\tau(s')
    \right)
    \monbind
    \lambda e'.\,\sem{e' : \tau}
    \tag{\text{by monad associativity}} \\
    &=
    \left(
      \symbigsemstep{s : \tau}{n}
      \monbind
      \mathsf{act}_\tau
    \right)
    \monbind
    \lambda e'.\,\sem{e' : \tau}
    \tag{\text{by eta-conversion}} \\
    &=
    \left(
      \mathsf{act}_\tau(s)
      \monbind
      \lambda e.\,\bigsemstep{e : \tau}{n}
    \right)
    \monbind
    \lambda e'.\,\sem{e' : \tau}
    \tag{\text{by induction hypothesis}} \\
    &=
    \mathsf{act}_\tau(s)
    \monbind
    \lambda e.\,
      \left(\bigsemstep{e : \tau}{n}
      \monbind
      \lambda e'.\,\sem{e' : \tau}\right)
    \tag{\text{by monad associativity}} \\
    &=
    \mathsf{act}_\tau(s)
    \monbind
    \lambda e.\,\bigsemstep{e : \tau}{n+1}.
    \tag{\text{by definition of n-step semantics}}
  \end{align}
\end{proof}

\begin{theorem}[Actual interpretation of initialized symbolic semantics]
  Let $e:\tau$ be closed and well typed. Then, for every
  $n\in\mathbb N$,
  \[
    \symbigsemstep{\mathsf{init}_\tau(e:\tau)}{n}
    \monbind
    \mathsf{act}_\tau
    =
    \bigsemstep{e:\tau}{n}.
  \]
\end{theorem}

\begin{proof}
  \begin{align}
    &\symbigsemstep{\mathsf{init}_\tau(e:\tau)}{n}
    \monbind
    \mathsf{act}_\tau
    \notag
    \\
    &=
    \mathsf{act}_\tau\bigl(\mathsf{init}_\tau(e:\tau)\bigr)
    \monbind
    \lambda e'.\,\bigsemstep{e':\tau}{n}
    \tag{\text{by \Cref{thm:actual-n-step}}}
    \\
    &=
    \delta_{e:\tau}
    \monbind
    \lambda e'.\,\bigsemstep{e':\tau}{n}
    \tag{\text{by the initial actual interpretation lemma}}
    \\
    &=
    \bigsemstep{e:\tau}{n}.
    \tag{\text{by monad left identity}}
  \end{align}
\end{proof}

\subsubsection{Expected interpretation of symbolic semantics}

The expected interpretation also starts from a symbolic state
$(\sigma \mid \hat e : \tau) \in State_\tau$, but replaces the symbolic variables in the residual expression $\hat e$ with the means of the samples recorded in $\sigma$, rather than sampling them.

First, we consider the sampling environment alone and define the following, replacing each symbolic sample in the sampling environment by the mean of its distribution. Distribution parameters are evaluated sequentially under the valuation induced by the preceding bindings:

\begin{align*}
  \sem{\emptyset}_{\mathsf{mean}}
  &=
  \delta_{()},\\
  \sem{\sigma,u\sim D(\vec a)}_{\mathsf{mean}}
  &=
  \sem{\sigma}_{\mathsf{mean}}
  \monbind
  \lambda\rho.\,
  \delta_{\rho\left[
  u\mapsto
  \mathbb{E}_{X\sim D(\vec r)}[X]
  \right]},
\end{align*}

with $\rho\vdash\vec a\Downarrow\vec r$. 

Note $\sem{\sigma}_{\mathsf{mean}}$ is a Dirac measure concentrated on a single valuation, where 
$\sem{\sigma}_{\mathsf{mean}} \in \mathcal{D}(\mathbb{R}^n)$.


\begin{example}
  \[
    \sem{
      u_1\sim\uniform(1,2),\;
      u_2\sim\uniform(0,u_1)
    }_{\mathsf{mean}}
    =
    \delta_{
      [u_1\mapsto 1.5,\;u_2\mapsto 0.75]
    }.
  \]
\end{example}

Now, we define a realization function.

\begin{definition}[Expected realization]
  Given a valuation $\rho$, the expected realization
  \[
    \mathsf{erealize}_\rho :
    \widehat{Expr_\tau}\to Expr_\tau
  \]
  is defined structurally by
  \begin{align*}
    \mathsf{erealize}_\rho(x)
    &=
    x,
    \\
    \mathsf{erealize}_\rho(\text{true})
    &=
    \text{true},
    \\
    \mathsf{erealize}_\rho(\text{false})
    &=
    \text{false},
    \\
    \mathsf{erealize}_\rho(\sym(a))
    &=
    r
    \quad \text{where }\rho\vdash a\Downarrow r,
    \\
    \mathsf{erealize}_\rho
      (\letkw\ x=\hat e_1\ \inkw\ \hat e_2)
    &=
    \letkw\ x=\mathsf{erealize}_\rho(\hat e_1)
    \ \inkw\ \mathsf{erealize}_\rho(\hat e_2),
    \\
    \mathsf{erealize}_\rho
      (\ifkw\ \hat e_1\ \thenkw\ \hat e_2\ \elsekw\ \hat e_3)
    &=
    \ifkw\ \mathsf{erealize}_\rho(\hat e_1)
    \ \thenkw\ \mathsf{erealize}_\rho(\hat e_2)
    \ \elsekw\ \mathsf{erealize}_\rho(\hat e_3),
    \\
    \mathsf{erealize}_\rho(\hat e_1+\hat e_2)
    &=
    \mathsf{erealize}_\rho(\hat e_1)
    +
    \mathsf{erealize}_\rho(\hat e_2),
    \\
    \mathsf{erealize}_\rho
      \bigl(\uniform[\E](\hat e_1,\hat e_2)\bigr)
    &=
    \mathsf{mean}_{\mathsf{Uniform}}
      \bigl(
        \mathsf{erealize}_\rho(\hat e_1),
        \mathsf{erealize}_\rho(\hat e_2)
      \bigr),
    \\
    \mathsf{erealize}_\rho
      \bigl(\uniform[\G](\hat e_1,\hat e_2)\bigr)
    &=
    \uniform\bigl(
      \mathsf{erealize}_\rho(\hat e_1),
      \mathsf{erealize}_\rho(\hat e_2)
    \bigr) \\
    &\ldots
  \end{align*}
  All other constructs are handled structurally.
\end{definition}

Strictly speaking, $\mathsf{erealize}_\rho$ acts on fully type-annotated symbolic residual expressions, preserving their type annotations.

\begin{lemma}\label{lem:erealize-lift}
  \[
    \mathsf{erealize}_{()}(\mathsf{lift}(e : \tau)) = \exptrans{e : \tau} : \tau
  \]
\end{lemma}

\begin{proof}
  By induction on $e : \tau$.

  \paragraph{Case $e=c:\Float m$.}
  \begin{align*}
    \mathsf{erealize}_{()}
    \bigl(\mathsf{lift}(c:\Float m)\bigr)
    &=
    \mathsf{erealize}_{()}
    \bigl(\sym(c):\Float m\bigr)
    \tag{by the definition of $\mathsf{lift}$}
    \\
    &=
    c:\Float m
    \tag{since $()\vdash c\Downarrow c$}
    \\
    &=
    \exptrans{c:\Float m}:\Float m
    \tag{by the definition of $\exptrans{\cdot}$}
  \end{align*}

  \paragraph{Case $e=e_1+e_2:\Float m$.}
  \begin{align*}
    \mathsf{erealize}_{()}
    \bigl(\mathsf{lift}(e_1+e_2)\bigr)
    &=
    \mathsf{erealize}_{()}
    \bigl(
      \mathsf{lift}(e_1)+\mathsf{lift}(e_2)
    \bigr)
    \tag{by the definition of $\mathsf{lift}$}
    \\
    &=
    \mathsf{erealize}_{()}\bigl(\mathsf{lift}(e_1)\bigr)
    +
    \mathsf{erealize}_{()}\bigl(\mathsf{lift}(e_2)\bigr)
    \tag{by the definition of $\mathsf{erealize}_{()}$}
    \\
    &=
    \exptrans{e_1}+\exptrans{e_2}
    \tag{by the I.H.}
    \\
    &=
    \exptrans{e_1+e_2}
    \tag{by the definition of $\exptrans{\cdot}$}
  \end{align*}

  \paragraph{Case $e=\uniform(e_1,e_2):\Float{\E}$.}
  \begin{align*}
    &\mathsf{erealize}_{()}
    \bigl(
      \mathsf{lift}
      (\uniform(e_1,e_2):\Float{\E})
    \bigr)
    \notag
    \\
    &=
    \mathsf{erealize}_{()}
    \bigl(
      \uniform[\E]
      (\mathsf{lift}(e_1),\mathsf{lift}(e_2))
      :\Float{\E}
    \bigr)
    \tag{by the definition of $\mathsf{lift}$}
    \\
    &=
    \mathsf{mean}_{\mathsf{Uniform}}
    \bigl(
      \mathsf{erealize}_{()}(\mathsf{lift}(e_1)),
      \mathsf{erealize}_{()}(\mathsf{lift}(e_2))
    \bigr)
    \tag{by the definition of $\mathsf{erealize}_{()}$}
    \\
    &=
    \mathsf{mean}_{\mathsf{Uniform}}
    \bigl(\exptrans{e_1},\exptrans{e_2}\bigr)
    \tag{by the I.H.}
    \\
    &=
    \exptrans{\uniform(e_1,e_2):\Float{\E}}:\Float{\E}
    \tag{by the definition of $\exptrans{\cdot}$}
  \end{align*}

  \paragraph{Case $e=\uniform(e_1,e_2):\Float{\G}$.}
  \begin{align*}
    &\mathsf{erealize}_{()}
    \bigl(
      \mathsf{lift}
      (\uniform(e_1,e_2):\Float{\G})
    \bigr)
    \notag
    \\
    &=
    \mathsf{erealize}_{()}
    \bigl(
      \uniform[\G]
      (\mathsf{lift}(e_1),\mathsf{lift}(e_2))
      :\Float{\G}
    \bigr)
    \tag{by the definition of $\mathsf{lift}$}
    \\
    &=
    \uniform\bigl(
      \mathsf{erealize}_{()}(\mathsf{lift}(e_1)),
      \mathsf{erealize}_{()}(\mathsf{lift}(e_2))
    \bigr)
    \tag{by the definition of $\mathsf{erealize}_{()}$}
    \\
    &=
    \uniform\bigl(\exptrans{e_1},\exptrans{e_2}\bigr)
    \tag{by the I.H.}
    \\
    &=
    \exptrans{\uniform(e_1,e_2):\Float{\G}}:\Float{\G}
    \tag{by the definition of $\exptrans{\cdot}$}
  \end{align*}

  \paragraph{All other cases analogous.}
\end{proof}


\begin{definition}[Single-state expected interpretation]
  The single-state expected interpretation
  \[
    \mathsf{exp}_\tau:
    State_\tau\to\mathcal{D}(Expr_\tau)
  \]
  is defined by

\end{lemma}

\begin{proof}
  \begin{align}

    &=
    \delta_{()}
    \monbind
    \lambda\rho.\,

    &=
    \delta_{\mathsf{erealize}_{()}(\mathsf{lift}(e:\tau)):\tau}
    \tag{by monad left identity}

    \delta_{\exptrans{e:\tau}:\tau}

\end{proof}

\begin{lemma}[Expectation of bind]\label{lem:expectation-of-bind}
  Let $\mu\in\mathcal D(X)$ be a probability measure, and let $k:X\to\mathcal D(\mathbb R)$ be a probability kernel. Assuming finite expectation,
  \[
    \mathbb E[\mu\monbind k]
    =
    \int_X
      \mathbb E[k(x)]
      \,d\mu(x).
  \]
\end{lemma}

\begin{proof}
  By definition of monadic bind, for every measurable \(A \subseteq \mathbb R\),
  \[
    (\mu \monbind k)(A)
    =
    \int_X k(x)(A)\,d\mu(x).
  \]
  Therefore, by the definition of expectation and Fubini's theorem,
  \begin{align}
    \mathbb E[\mu\monbind k]
    &=
    \int_{\mathbb R} r\,d(\mu\monbind k)(r)
    \tag{\text{by the definition of expectation}}
    \\
    &=
    \int_X
    \left(
      \int_{\mathbb R} r\,dk(x)(r)
    \right)
    d\mu(x)
    \tag{\text{by the definition of monadic bind and Fubini's theorem}}
    \\
    &=
    \int_X \mathbb E[k(x)]\,d\mu(x)
    \tag{\text{by the definition of expectation}}.
  \end{align}
\end{proof}


\begin{theorem}[Soundness $n$-step]

  Let $e : \tau$ be a well-typed expression, where $\tau = \Float{m}$ and $m = \G$ or $\E$. Then, for every $n$, when restricted to values:
  \[
    \mathbb E\!\left[
      \left.\bigsemstep{e : \tau}{n}\right|_{Val_\tau}
    \right]
    =
    \mathbb E\!\left[
      \left.\bigsemstep{\exptrans{e : \tau} : \tau}{n}\right|_{Val_\tau}
    \right].

  % and in particular:
  % \[
  %   \bigsem{e : \Float{\G}}
  %   =
  %   \bigsem{\exptrans{e : \Float{\G}} : \Float{\G}}.
  % \]
\end{theorem}

\begin{proof}
  Let
  \[
    \mu_n
    =
    \symbigsemstep{\mathsf{init}_\tau(e:\tau)}{n}.
  \]
  Then
  \begin{align}
    \mathbb E\!\left[
      \left.\bigsemstep{e:\tau}{n}\right|_{Val_\tau}
    \right]
    &=
    \mathbb E\!\left[
      \left.(\mu_n\monbind\mathsf{act}_\tau)\right|_{Val_\tau}
    \right]
    \tag{\text{by the actual interpretation theorem}}
    \\
    &=
    \int
      \mathbb E\!\left[
        \left.\mathsf{act}_\tau(s)\right|_{Val_\tau}
      \right]
      \,d\mu_n(s)
    \tag{by~\Cref{lem:expectation-of-bind}}
    \\
    &=
    \int
      \mathbb E\!\left[
        \left.\mathsf{exp}_\tau(s)\right|_{Val_\tau}
      \right]
      \,d\mu_n(s)
    \tag{by~\Cref{thm:agreement-of-interpretations}}
    \\
    &=
    \mathbb E\!\left[
      \left.(\mu_n\monbind\mathsf{exp}_\tau)\right|_{Val_\tau}
    \right]
    \tag{by~\Cref{lem:expectation-of-bind}}
    \\
    &=
    \mathbb E\!\left[
      \left.
        \bigsemstep{\exptrans{e:\tau}:\tau}{n}
      \right|_{Val_\tau}
    \right].
    \tag{\text{by the expected interpretation theorem}}
  \end{align}
\end{proof}


\begin{lemma}[Expectation commutes with increasing limits]
\label{lem:expectation-increasing-limits}
Let \(\mu_0 \leq \mu_1 \leq \mu_2 \leq \cdots\) be an increasing sequence of subprobability measures over \(\mathbb{R}_{\geq 0}\), and suppose
\[
\mu = \lim_{n \to \infty} \mu_n,
\]
where the limit is taken setwise, i.e., \(\mu(A) = \lim_{n \to \infty} \mu_n(A)\) for every measurable set \(A\). Then
\[
\mathbb{E}[\mu] = \lim_{n \to \infty} \mathbb{E}[\mu_n].
\]
\end{lemma}

\begin{proof}
By definition,
\[
\mathbb{E}[\mu] = \int_{\mathbb{R}_{\geq 0}} x \, d\mu(x).
\]
Since \((\mu_n)_{n \in \mathbb{N}}\) is increasing and \(\mu = \lim_{n \to \infty} \mu_n\), continuity of integration with respect to increasing limits of measures gives
\[
\int_{\mathbb{R}_{\geq 0}} x \, d\mu(x)
=
\lim_{n \to \infty}
\int_{\mathbb{R}_{\geq 0}} x \, d\mu_n(x).
\]
Hence,
\[
\mathbb{E}[\mu] = \lim_{n \to \infty} \mathbb{E}[\mu_n].
\]
\end{proof}


\begin{theorem}[Soundness big step]
    Let $e : \tau$ be a well-typed expression, where $\tau = \Float{m}$ and $m = \G$ or $\E$. Then:
  \[
    \mathbb{E}[\bigsem{e : \tau}]
    =
    \mathbb{E}[\bigsem{\exptrans{e : \tau} : \tau}].
  \]
\end{theorem}

\begin{proof}
  For each $n\in\mathbb N$, let
  \[
    \mu_n
    =
    \left.
      \bigsemstep{e:\tau}{n}
    \right|_{Val_\tau}
    \qquad\text{and}\qquad
    \nu_n
    =
    \left.
      \bigsemstep{\exptrans{e:\tau}:\tau}{n}
    \right|_{Val_\tau}.
  \]
  Since values are absorbing, $(\mu_n)_{n\in\mathbb N}$ and
  $(\nu_n)_{n\in\mathbb N}$ are increasing sequences of measures on
  $Val_\tau$. 
  \begin{align}
    \mathbb E\!\left[\bigsem{e:\tau}\right]
    &=
    \mathbb E\!\left[\lim_{n\to\infty}\mu_n\right]
    \tag{\text{by the definition of the big-step semantics}}
    \\
    &=
    \lim_{n\to\infty}
    \mathbb E\!\left[\mu_n\right]
    \tag{\text{by \Cref{lem:expectation-increasing-limits}}}
    \\
    &=
    \lim_{n\to\infty}
    \mathbb E\!\left[\nu_n\right]
    \tag{\text{by \Cref{thm:soundness-n-step}}}
    \\
    &=
    \mathbb E\!\left[\lim_{n\to\infty}\nu_n\right]
    \tag{\text{by \Cref{lem:expectation-increasing-limits}}}
    \\
    &=
    \mathbb E\!\left[
      \bigsem{\exptrans{e:\tau}:\tau}
    \right].
    \tag{\text{by the definition of the big-step semantics}}
  \end{align}
\end{proof}
