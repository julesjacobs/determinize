\begin{lemma}[Type preservation]
  Let $e$ be a well-typed expression of type $\tau$, and assume that
  $\{e' \mid \emptyset \vdash e' : \tau'\}$ is measurable for every type $\tau'$.
  Then
  \[
    \sem{e : \tau}\bigl(\{\,e' \mid \emptyset \nvdash e' : \tau\,\}\bigr) = 0.
  \]
  That is, the small-step semantics assigns measure zero to the set of
  expressions that are not well-typed at type $\tau$.
\end{lemma}

\begin{lemma}[Step is a Markov kernel]
  The map $e \mapsto \sem{e : \tau}$ defines a Markov kernel $Expr_\tau \to \mathcal{D}(Expr_\tau)$.
\end{lemma}

\subsection{Big-step}

\begin{lemma}[$n$-step semantics]
  The $n$-step semantics $\bigsemstep{e : \tau}{n}$ is defined by iterating the small-step kernel $n$ times, and is itself a Markov kernel:

  \[
    \bigsemstep{e : \tau}{n} =   
    \begin{cases}
      \delta_{e : \tau} &\text{ if } n = 0 \\
      \bigsemstep{e : \tau}{n-1} \monbind \lambda e'.\sem{e' : \tau} &\text{ if } n > 0.
    \end{cases}
  \]

\end{lemma}

\begin{lemma}[Values are absorbing]\label{lem:values-absorbing}
  If $v : \tau \in Val_\tau$, then
  \[
    \sem{v : \tau} = \delta_{v : \tau}
    \qquad\text{and}\qquad
    \bigsemstep{v : \tau}{n} = \delta_{v : \tau}
  \]
  for every $n \in \mathbb{N}$.
\end{lemma}

\begin{proof}
  The first equality follows by inspection of values in
  \Cref{tab:small-step-semantics}. The second equality follows by induction on
  $n$, using the first equality and the left-unit law for Dirac measures in the
  successor case.
\end{proof}

\begin{lemma}[Monotone accumulation of mass]
  For all $n \in \mathbb{N}$:
  \begin{enumerate}
    \item $\bigsemstep{e : \tau}{n}(Val_\tau) \leq \bigsemstep{e : \tau}{n+1}(Val_\tau)$
  \end{enumerate}
  That is, probability mass on values accumulates monotonically.
\end{lemma}

\begin{lemma}[Big-step semantics exists]
  The pointwise limit
  \[
    \bigsem{e : \tau}(A)
    := \lim_{n \to \infty} \bigsemstep{e : \tau}{n}(A)
  \]
  exists for every measurable $A \subseteq Val_{\tau}$, by the monotone convergence theorem.
\end{lemma}

\subsection{Symbolic Expressions}
A symbolic state is:
\[
  \hat{s} = (\sigma \mid \hat{e} : \tau) \in State_\tau.
\]

where $\sigma$ is the symbolic sample enviornment and $\hat{e} : \tau \in \widehat{Expr_\tau}$ is the well-typed symbolic residual expression.

The syntax for $\hat{e}$ is defined as follows:
\begin{figure}[h]
	\begin{align*}
    a ::= &\; c \mid u \mid a_1 + a_2 \mid c \cdot a & \text{affine expressions} \\
		\hat e ::= &\; x \mid \text{true} \mid \text{false} \mid () \mid \sym(a) & \text{constants} \\
		| &\; \letkw \; x = \hat e_1\; \inkw \; \hat e_2  & \text{let binding} \\
	    | &\; \hat e_1 + \hat e_2 \mid  \hat e_1 - \hat e_2 \mid \hat e_1 \times \hat e_2 \mid \hat e_1\; /\; \hat e_2 & \text{arithmetic} \\
	    | &\; \hat e_1 < \hat e_2  & \text{comparisons} \\
      | &\; \uniform[m](\hat e_1,\hat e_2) & \text{distributions, } m \in \{\E,\G\} \\
	    | &\; \ifkw\; \hat e_1\; \thenkw\; \hat e_2\; \elsekw\; \hat e_3 & \text{conditionals} \\
    | &\; \langle \hat e_1,\hat e_2\rangle & \text{pairs} \\
    \ldots & \text{etc.}
	\end{align*}
\label{fig:grammar}
\end{figure}

And formally, a symbolic sample environment $\sigma$ is an ordered context:
\[
  \sigma
  =
  \bigl(
    u_1 \sim D_1(\vec a_1),\;
    \ldots,\;
    u_k \sim D_k(\vec a_k)
  \bigr).
\]
Here each \(\vec a_i\) is a finite tuple of parameters:
\[
  \vec a_i = (a_{i,1},\ldots,a_{i,n_i})
  \qquad\text{with}\qquad
  a_{i,j} \in \mathsf{Aff}(\{u_1,\ldots,u_{i-1}\}).
\]
Thus each distribution may take any number of parameters, and every parameter is
affine in the samples that appear earlier in \(\sigma\). For \(i=1\), this means
the parameters are constants.

\subsubsection{Type System}

In a given symbolic state, only the symbolic residual expression is typed. The type system has a single judgment form:
\[
  \Gamma \vdash \hat e : \tau.
\]

\paragraph{Symbolic residual expressions.}
The ordinary typing rules lift to symbolic residual expressions. 
\begin{mathpar}
  \inferrule[\textsc{Var}]
    {x:\tau \in \Gamma}
    {\Gamma \vdash x : \tau}
  \and
  \inferrule[\textsc{Sym}]
    {a\ \text{is affine}}
    {\Gamma \vdash \sym(a) : \Float{m}}
  \and
  \inferrule[\textsc{Let}]
    {\Gamma \vdash \hat e_1 : \tau_1
      \\ \Gamma,x:\tau_1 \vdash \hat e_2 : \tau_2}
    {\Gamma \vdash
      \letkw\;x=\hat e_1\;\inkw\;\hat e_2 : \tau_2}
  \and
  \inferrule[\textsc{Pair}]
    {\Gamma \vdash \hat e_1 : \tau_1
      \\ \Gamma \vdash \hat e_2 : \tau_2}
    {\Gamma \vdash
      \langle \hat e_1,\hat e_2\rangle : \tau_1 \times \tau_2}
  \and
  \inferrule[\textsc{Add}]
    {\Gamma \vdash \hat e_1 : \Float{m_1}
      \\ \Gamma \vdash \hat e_2 : \Float{m_2}
      \\ m_1 \preceq m
      \\ m_2 \preceq m}
    {\Gamma \vdash \hat e_1 + \hat e_2 : \Float{m}}
  \and
  \inferrule[\textsc{Uniform}]
    {\Gamma \vdash \hat e_1 : \Float{m_1}
      \\ \Gamma \vdash \hat e_2 : \Float{m_2}
      \\ m_1 \preceq m
      \\ m_2 \preceq m}
    {\Gamma \vdash \uniform[m](\hat e_1,\hat e_2) : \Float{m}}
  \and
  \inferrule[\textsc{Less}]
    {\Gamma \vdash \hat e_1 : \Float{m_1}
      \\ \Gamma \vdash \hat e_2 : \Float{m_2}
      \\ m_1 \preceq \G
      \\ m_2 \preceq \G}
    {\Gamma \vdash \hat e_1 < \hat e_2 : \Bool}
\end{mathpar}

\subsubsection{Small-step symbolic semantics}
We start with the following notation for the one-step symbolic semantics. The small-step symbolic semantics is given in~\Cref{tab:symbolic-small-step-semantics}.

\[
  \widehat{\sem{\hat{s} : \tau}} \in \mathcal{D}(State_\tau)
\]

Now, define the $n$-step symbolic semantics as follows:

\begin{align*}
  \widehat{\bigsemstep{\hat{s} : \tau}{n}} =
  \begin{cases}
    \delta_{\hat{s} : \tau} &\text{ if } n = 0 \\
    \widehat{\bigsemstep{\hat{s} : \tau}{n-1}} \monbind \lambda \hat{s}. \widehat{\sem{\hat{s} : \tau}} &\text{ if } n > 0
  \end{cases}
\end{align*}

where $\widehat{\bigsemstep{\hat{s} : \tau}{n}} \in \mathcal{D}(State_\tau)$.

\subsubsection{Actual interpretation of symbolic semantics}

The actual interpretation turns a symbolic state back into a distribution over
ordinary expressions by sampling the symbolic environment and substituting the
sampled values for the symbolic variables.  Suppose
\[
  \sigma =
  u_1 \sim D_1(\vec a_1),\ldots,
  u_n \sim D_n(\vec a_n),
  \qquad
  a_{i,j} \in \mathsf{Aff}(\{u_1,\ldots,u_{i-1}\}).
\]
Although \(\sigma\) is an ordered context, it determines a distribution over
tuples of real numbers:
\[
  \sem{\sigma}_{\mathsf{env}} \in \mathcal{D}(\mathbb{R}^n).
\]

This distribution is defined sequentially by:
\[
\sem{\sigma}_{\mathsf{env}} =
\begin{cases}
  \delta_{()}
    & \text{if } \sigma = \emptyset, \\[0.5em]

  D(\vec a)
  \monbind
  \lambda r_1.\,
  \sem{\sigma'[r_1/u]}_{\mathsf{env}}
  \monbind
  \lambda(r_2,\ldots,r_n).\,
  \delta_{(r_1,r_2,\ldots,r_n)}
    & \text{if } \sigma = u \sim D(\vec a), \sigma'.
\end{cases}
\]

\begin{example}
  Let's say:
  \[
    \sigma = u \sim \mathsf{Uniform}(0,1),\;
            v \sim \mathsf{Gaussian}(u,1),
  \]

  Then this gives:
  \[
    \sem{\sigma}_{\mathsf{env}}
    =
    \mathsf{Uniform}(0,1)
    \monbind
    \lambda r.\,
    \mathsf{Gaussian}(r,1)
    \monbind
    \lambda s.\,
    \delta_{(r,s)}.
  \]
\end{example}

For a symbolic residual expression \(\hat e\) and a tuple
\(\rho = (r_1,\ldots,r_n)\), write
\[
  \mathsf{realize}_\rho(\hat e)
\]
for the ordinary expression obtained by replacing each occurrence of
\(\sym(a)\) in \(\hat e\) with the real number obtained by evaluating the
affine expression \(a\) under \(\rho\):
\[
  \sym(a)
  \mapsto
  a[\rho]
  =
  a[r_1/u_1,\ldots,r_n/u_n].
\]

If
\[
  \vdash \hat e : \tau,
\]
then
\[
  \mathsf{realize}_\rho(\hat e) \in Expr_\tau.
\]
We also extend realization to symbolic evaluation contexts.  For a symbolic
context \(C[\cdot]\), \(\mathsf{realize}_\rho(C)[e]\) means: realize the
symbolic expressions in \(C[\cdot]\) while leaving its hole open, then fill that
hole with the expression \(e\).  

\begin{lemma}[Realization commutes with plugging into contexts]
  \[
    \mathsf{realize}_\rho(C[\hat e])
    =
    \mathsf{realize}_\rho(C)[\mathsf{realize}_\rho(\hat e)].
  \]
\end{lemma}

\begin{proof}
  By induction on the structure of \(C[\cdot]\).

  If \(C[\cdot]=[\cdot]\), then
  \[
    \mathsf{realize}_\rho(C[\hat e])
    =
    \mathsf{realize}_\rho(\hat e)
    =
    \mathsf{realize}_\rho(C)[\mathsf{realize}_\rho(\hat e)].
  \]

  For the let context case, suppose
  \[
    C[\cdot]
    =
    \letkw\ x=C_1[\cdot]\ \inkw\ \hat e_2.
  \]
  Then
  \begin{align}
    \mathsf{realize}_\rho(C[\hat e])
    &=
    \letkw\ x=\mathsf{realize}_\rho(C_1[\hat e])
    \ \inkw\ \mathsf{realize}_\rho(\hat e_2)
    \tag{\text{by definition of realization}} \\
    &=
    \letkw\ x=
      \mathsf{realize}_\rho(C_1)[\mathsf{realize}_\rho(\hat e)]
    \ \inkw\ \mathsf{realize}_\rho(\hat e_2)
    \tag{\text{by induction hypothesis}} \\
    &=
    \mathsf{realize}_\rho(C)[\mathsf{realize}_\rho(\hat e)].
    \tag{\text{by definition of context realization}}
  \end{align}
  Other context forms are analogous.
\end{proof}

\begin{lemma}[Realization substitution]
  \[
    \mathsf{realize}_\rho(\hat e[\hat v/x])
    =
    \mathsf{realize}_\rho(\hat e)
    [\mathsf{realize}_\rho(\hat v)/x].
  \]
\end{lemma}

\begin{definition}[Single-state actual interpretation]
  The single-state actual interpretation is the function:
  \[
    \mathsf{act}_\tau : State_\tau \to \mathcal{D}(Expr_\tau)
  \]
  defined by
  \[
  \begin{aligned}
    \mathsf{act}_\tau(\sigma \mid \hat e)
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
    \delta_{\mathsf{realize}_\rho(\hat e) : \tau}.
  \end{aligned}
  \]
\end{definition}

\begin{definition}[Lifted actual interpretation]
  The lifted actual interpretation acts on distributions over symbolic states:
  \[
    \mathsf{Act}_\tau :
    \mathcal{D}(State_\tau) \to \mathcal{D}(Expr_\tau).
  \]
  It is defined by monadic bind against the single-state kernel:
  \[
  \begin{aligned}
    \mathsf{Act}_\tau(\mu)
    &=
    \mu \monbind \mathsf{act}_\tau \\
    &=
    \mu \monbind
    \lambda(\sigma \mid \hat e).\,\mathsf{act}_\tau(\sigma \mid \hat e).
  \end{aligned}
  \]
\end{definition}

\begin{theorem}[Actual interpretation of symbolic \(n\)-step semantics]
  If \(\hat{s} = (\sigma \mid \hat e : \tau) \in State_\tau\), then for
  every \(n \in \mathbb{N}\),
  \[
    \mathsf{Act}_\tau\!\left(
      \widehat{\bigsemstep{\hat{s} : \tau}{n}}
    \right)
    =
    \mathsf{act}_\tau(\hat{s})
    \monbind
    \lambda e.\,\bigsemstep{e : \tau}{n}.
  \]
\end{theorem}

\begin{proof}
  The proof is by induction on \(n\), with the statement quantified over all
  well-typed symbolic states.

  For \(n=0\), let \(\hat{s} \in State_\tau\). Then
  \begin{align}
    \mathsf{Act}_\tau\!\left(
      \widehat{\bigsemstep{\hat{s} : \tau}{0}}
    \right)
    &=
    \mathsf{Act}_\tau(\delta_{\hat{s} : \tau})
    \tag{\text{by definition of n-step symbolic semantics}} \\
    &=
    \delta_{\hat{s} : \tau}
    \monbind
    \mathsf{act}_\tau
    \tag{\text{by definition of Act}} \\
    &=
    \mathsf{act}_\tau(\hat{s})
    \tag{\text{by monad left identity}} \\
    &=
    \mathsf{act}_\tau(\hat{s})
    \monbind
    \lambda e.\,\delta_{e : \tau}
    \tag{\text{by monad right identity}} \\
    &=
    \mathsf{act}_\tau(\hat{s})
    \monbind
    \lambda e.\,\bigsemstep{e : \tau}{0}.
    \tag{\text{by definition of n-step semantics}}
  \end{align}

  For the inductive step, assume the statement holds for \(n\) for every
  well-typed symbolic state, and let
  \(\hat{s} = (\sigma \mid \hat e : \tau) \in State_\tau\). We use the
  one-step compatibility fact:

  \begin{equation}
    \mathsf{Act}_\tau\!\left(\widehat{\sem{\hat{s} : \tau}}\right)
    =
    \mathsf{act}_\tau(\hat{s})
    \monbind
    \lambda e'.\,\sem{e' : \tau}
  \end{equation}
  for every well-typed symbolic state \(\hat{s} \in State_\tau\). Then
  \begin{align}
    &\mathsf{Act}_\tau\!\left(
      \widehat{\bigsemstep{\hat{s} : \tau}{n+1}}
    \right) \notag \\
    &=
    \mathsf{Act}_\tau\!\left(
      \widehat{\bigsemstep{\hat{s} : \tau}{n}}
      \monbind
      \lambda \hat{s}'.\,\widehat{\sem{\hat{s}' : \tau}}
    \right)
    \tag{\text{by definition of symbolic n-step}} \\
    &=
    \left(
      \widehat{\bigsemstep{\hat{s} : \tau}{n}}
      \monbind
      \lambda \hat{s}'.\,\widehat{\sem{\hat{s}' : \tau}}
    \right)
    \monbind
    \mathsf{act}_\tau
    \tag{\text{by definition of Act}} \\
    &=
    \widehat{\bigsemstep{\hat{s} : \tau}{n}}
    \monbind
    \lambda \hat{s}'.\,
      \left(\widehat{\sem{\hat{s}' : \tau}}
      \monbind \mathsf{act}_\tau\right)
    \tag{\text{by monad associativity}} \\
    &=
    \widehat{\bigsemstep{\hat{s} : \tau}{n}}
    \monbind
    \lambda \hat{s}'.\,
      \mathsf{Act}_\tau\!\left(\widehat{\sem{\hat{s}' : \tau}}\right)
    \tag{\text{by definition of Act}} \\
    &=
    \widehat{\bigsemstep{\hat{s} : \tau}{n}}
    \monbind
    \lambda \hat{s}'.\,
      \left(\mathsf{act}_\tau(\hat{s}')
      \monbind \lambda e'.\,\sem{e' : \tau}\right)
    \tag{\text{by one-step compatibility}} \\
    &=
    \left(
      \widehat{\bigsemstep{\hat{s} : \tau}{n}}
      \monbind
      \lambda \hat{s}'.\,\mathsf{act}_\tau(\hat{s}')
    \right)
    \monbind
    \lambda e'.\,\sem{e' : \tau}
    \tag{\text{by monad associativity}} \\
    &=
    \mathsf{Act}_\tau\!\left(
      \widehat{\bigsemstep{\hat{s} : \tau}{n}}
    \right)
    \monbind
    \lambda e'.\,\sem{e' : \tau}
    \tag{\text{by definition of Act}} \\
    &=
    \left(
      \mathsf{act}_\tau(\hat{s})
      \monbind
      \lambda e.\,\bigsemstep{e : \tau}{n}
    \right)
    \monbind
    \lambda e'.\,\sem{e' : \tau}
    \tag{\text{by induction hypothesis}} \\
    &=
    \mathsf{act}_\tau(\hat{s})
    \monbind
    \lambda e.\,
      \left(\bigsemstep{e : \tau}{n}
      \monbind
      \lambda e'.\,\sem{e' : \tau}\right)
    \tag{\text{by monad associativity}} \\
    &=
    \mathsf{act}_\tau(\hat{s})
    \monbind
    \lambda e.\,\bigsemstep{e : \tau}{n+1}.
    \tag{\text{by definition of n-step semantics}}
  \end{align}
\end{proof}

\begin{lemma}[Strengthened one-step compatibility]
  Let \(C[\cdot]\) be a well-typed symbolic evaluation context whose hole has
  type \(\tau_0\) and whose whole expression has type \(\tau\).  If
  \((\sigma \mid \hat e_0 : \tau_0) \in State_{\tau_0}\), then
  \[
  \begin{aligned}
    &
    \widehat{\sem{(\sigma \mid \hat e_0 : \tau_0)}}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
      \mathsf{act}_\tau(\sigma' \mid C[\hat g]) \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \left(
        \sem{\mathsf{realize}_\rho(\hat e_0) : \tau_0}
        \monbind
        \lambda g.\,
          \delta_{\mathsf{realize}_\rho(C)[g] : \tau}
      \right).
  \end{aligned}
  \]
\end{lemma}

\begin{proof}
  The proof is by induction on \(\hat e_0\).

  \paragraph{Value case.}
  \begin{align}
    &\widehat{\sem{(\sigma \mid \hat v : \tau_0)}}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
      \mathsf{act}_\tau(\sigma' \mid C[\hat g]) \notag \\
    &=
    \delta_{(\sigma \mid \hat v : \tau_0)}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
      \mathsf{act}_\tau(\sigma' \mid C[\hat g])
    \tag{\text{by symbolic value rule}} \\
    &=
    \mathsf{act}_\tau(\sigma \mid C[\hat v])
    \tag{\text{by monad left identity}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \delta_{\mathsf{realize}_\rho(C[\hat v]) : \tau}
    \tag{\text{by definition of act}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \delta_{\mathsf{realize}_\rho(C)
        [\mathsf{realize}_\rho(\hat v)] : \tau}
    \tag{\text{by realization of contexts}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \left(
        \delta_{\mathsf{realize}_\rho(\hat v) : \tau_0}
        \monbind
        \lambda g.\,
          \delta_{\mathsf{realize}_\rho(C)[g] : \tau}
      \right)
    \tag{\text{by monad left identity}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \left(
        \sem{\mathsf{realize}_\rho(\hat v) : \tau_0}
        \monbind
        \lambda g.\,
          \delta_{\mathsf{realize}_\rho(C)[g] : \tau}
      \right).
    \tag{\text{by ordinary value rule}}
  \end{align}

  \paragraph{Let value case.}
  \begin{align}
    &\widehat{\sem{(\sigma \mid
      \letkw\ x=\hat v\ \inkw\ \hat e_2 : \tau_2)}}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
      \mathsf{act}_\tau(\sigma' \mid C[\hat g]) \notag \\
    &=
    \delta_{(\sigma \mid \hat e_2[\hat v/x] : \tau_2)}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
      \mathsf{act}_\tau(\sigma' \mid C[\hat g])
    \tag{\text{by symbolic let rule}} \\
    &=
    \mathsf{act}_\tau(\sigma \mid C[\hat e_2[\hat v/x]])
    \tag{\text{by monad left identity}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \delta_{\mathsf{realize}_\rho(C[\hat e_2[\hat v/x]]) : \tau}
    \tag{\text{by definition of act}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \delta_{\mathsf{realize}_\rho(C)
        [\mathsf{realize}_\rho(\hat e_2[\hat v/x])] : \tau}
    \tag{\text{by realization of contexts}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \delta_{\mathsf{realize}_\rho(C)
        [\mathsf{realize}_\rho(\hat e_2)
          [\mathsf{realize}_\rho(\hat v)/x]] : \tau}
    \tag{\text{by realization commuting with substitution}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \left(
        \delta_{\mathsf{realize}_\rho(\hat e_2)
          [\mathsf{realize}_\rho(\hat v)/x] : \tau_2}
        \monbind
        \lambda g.\,
          \delta_{\mathsf{realize}_\rho(C)[g] : \tau}
      \right)
    \tag{\text{by monad left identity}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \left(
        \sem{
          \letkw\ x=\mathsf{realize}_\rho(\hat v)
          \ \inkw\ \mathsf{realize}_\rho(\hat e_2) : \tau_2
        }
        \monbind
        \lambda g.\,
          \delta_{\mathsf{realize}_\rho(C)[g] : \tau}
      \right).
    \tag{\text{by ordinary let-value rule}}
  \end{align}

  \paragraph{Let non-value case.}
  \begin{align}
    &\widehat{\sem{(\sigma \mid \hat e_0 : \tau_2)}}
    \monbind
    \lambda(\sigma'' \mid \hat h).\,
      \mathsf{act}_\tau(\sigma'' \mid C[\hat h]) \notag \\
    &=
    \left(
      \widehat{\sem{(\sigma \mid \hat e_1 : \tau_1)}}
      \monbind
      \lambda(\sigma' \mid \hat g_1).\,
        \delta_{(\sigma' \mid
          \letkw\ x=\hat g_1\ \inkw\ \hat e_2) : \tau_2}
    \right)
    \monbind
    \lambda(\sigma'' \mid \hat h).\,
      \mathsf{act}_\tau(\sigma'' \mid C[\hat h])
    \tag{\text{by the symbolic let rule}} \\
    &=
    \widehat{\sem{(\sigma \mid \hat e_1 : \tau_1)}}
    \monbind
    \lambda(\sigma' \mid \hat g_1).\,
      \left(
        \delta_{(\sigma' \mid
          \letkw\ x=\hat g_1\ \inkw\ \hat e_2) : \tau_2}
        \monbind
        \lambda(\sigma'' \mid \hat h).\,
          \mathsf{act}_\tau(\sigma'' \mid C[\hat h])
      \right)
    \tag{\text{by associativity}} \\
    &=
    \widehat{\sem{(\sigma \mid \hat e_1 : \tau_1)}}
    \monbind
    \lambda(\sigma' \mid \hat g_1).\,
      \mathsf{act}_\tau(\sigma' \mid
        C[\letkw\ x=\hat g_1\ \inkw\ \hat e_2]).
    \tag{\text{by monad left identity}}
  \end{align}
  Let
  \[
    C_{\mathsf{let}}[\cdot]
    =
    C[\letkw\ x=[\cdot]\ \inkw\ \hat e_2].
  \]
  Hence:
  \begin{align}
    &\widehat{\sem{(\sigma \mid \hat e_1 : \tau_1)}}
    \monbind
    \lambda(\sigma' \mid \hat g_1).\,
      \mathsf{act}_\tau(\sigma' \mid
        C[\letkw\ x=\hat g_1\ \inkw\ \hat e_2]) \notag \\
    &=
    \widehat{\sem{(\sigma \mid \hat e_1 : \tau_1)}}
    \monbind
    \lambda(\sigma' \mid \hat g_1).\,
      \mathsf{act}_\tau(\sigma' \mid C_{\mathsf{let}}[\hat g_1])
    \tag{\text{by definition of \(C_{\mathsf{let}}\)}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \left(
        \sem{\mathsf{realize}_\rho(\hat e_1) : \tau_1}
        \monbind
        \lambda g.\,
          \delta_{\mathsf{realize}_\rho(C_{\mathsf{let}})[g] : \tau}
      \right)
    \tag{\text{by induction hypothesis}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \left(
        \sem{\mathsf{realize}_\rho(\hat e_1) : \tau_1}
        \monbind
        \lambda g.\,
          \delta_{\mathsf{realize}_\rho(C)
            [\letkw\ x=g\ \inkw\ \mathsf{realize}_\rho(\hat e_2)] : \tau}
      \right)
    \tag{\text{by realization of \(C_{\mathsf{let}}\)}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \left(
        \sem{
          \letkw\ x=\mathsf{realize}_\rho(\hat e_1)
          \ \inkw\ \mathsf{realize}_\rho(\hat e_2) : \tau_2
        }
        \monbind
        \lambda g.\,
          \delta_{\mathsf{realize}_\rho(C)[g] : \tau}
      \right).
    \tag{\text{by ordinary let rule}}
  \end{align}
\end{proof}

\begin{lemma}[One-step compatibility]
  If \(\hat{s} \in State_\tau\), then
  \begin{equation}
    \mathsf{Act}_\tau\!\left(\widehat{\sem{\hat{s} : \tau}}\right)
    =
    \mathsf{act}_\tau(\hat{s})
    \monbind
    \lambda e'.\,\sem{e' : \tau}
  \end{equation}
\end{lemma}

\begin{proof}
  Let \(\hat{s}=(\sigma \mid \hat e : \tau)\). Apply the strengthened one-step
  compatibility lemma with the empty surrounding expression \(C[\cdot]=[\cdot]\):
  \begin{align}
    &\mathsf{Act}_\tau\!\left(\widehat{\sem{\hat{s} : \tau}}\right) \notag \\
    &=
    \widehat{\sem{(\sigma \mid \hat e : \tau)}}
    \monbind
    \lambda(\sigma' \mid \hat g).\,
      \mathsf{act}_\tau(\sigma' \mid \hat g)
    \tag{\text{by definition of Act}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \left(
        \sem{\mathsf{realize}_\rho(\hat e) : \tau}
        \monbind
        \lambda g.\,\delta_{g : \tau}
      \right)
    \tag{\text{by strengthened one-step compatibility}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \sem{\mathsf{realize}_\rho(\hat e) : \tau}
    \tag{\text{by monad right identity}} \\
    &=
    \sem{\sigma}_{\mathsf{env}}
    \monbind
    \lambda\rho.\,
      \left(
        \delta_{\mathsf{realize}_\rho(\hat e) : \tau}
        \monbind
        \lambda e'.\,\sem{e' : \tau}
      \right)
    \tag{\text{by monad left identity}} \\
    &=
    \left(
      \sem{\sigma}_{\mathsf{env}}
      \monbind
      \lambda\rho.\,
        \delta_{\mathsf{realize}_\rho(\hat e) : \tau}
    \right)
    \monbind
    \lambda e'.\,\sem{e' : \tau}
    \tag{\text{by associativity}} \\
    &=
    \mathsf{act}_\tau(\hat{s})
    \monbind
    \lambda e'.\,\sem{e' : \tau}.
    \tag{\text{by definition of act}}
  \end{align}
\end{proof}
