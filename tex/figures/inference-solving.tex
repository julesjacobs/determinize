\textbf{(a) Subtype constraint reduction}\leanref{subtype-constraint-reduction}
\[
\begin{gathered}
\tyR a<:\tyR b
  \;\leadsto\;\{a\le b\}
  \qquad
\tyL\tau<:\tyL\sigma
  \;\leadsto\;\{\tau<:\sigma\}
  \qquad
\chi<:\chi
  \;\leadsto\;\varnothing
  \quad(\chi\in\{\tyU,\tyB\})\\[2pt]
(\tau_1\to\tau_2)<:(\sigma_1\to\sigma_2)
  \;\leadsto\;\{\sigma_1<:\tau_1,\;\tau_2<:\sigma_2\}\\[2pt]
(\tau_1\star\tau_2)<:(\sigma_1\star\sigma_2)
  \;\leadsto\;\{\tau_1<:\sigma_1,\;\tau_2<:\sigma_2\}
  \quad(\star\in\{\times,+\}).
\end{gathered}
\]

\smallskip
\textbf{(b) Type variable resolution}\leanref{type-shape-unification}\leanref{type-affinity-decoration}

\begin{minipage}[t]{0.36\linewidth}
\vspace{0pt}
\[
\begin{aligned}
h(\tyR a)&=\tyR b,\\
h(\tau\to\sigma)&=U\to V,\\
h(\tyL\tau)&=\tyL U,\\
h(\tau\star\sigma)&=U\star V,\\
h(\chi)&=\chi.
\end{aligned}
\]
\end{minipage}\hfill
\begin{minipage}[t]{0.62\linewidth}
\vspace{0pt}
\[
\begin{aligned}
X<:X
  &\;\leadsto\;\varnothing,\\[3pt]
X<:Y\ (X\ne Y)
  &\;\leadsto\;\text{defer }X<:Y,\\[3pt]
X<:\tau
  &\;\leadsto\;\{h(\tau)<:\tau\}
    \quad[X\mapsto h(\tau)],\\[3pt]
\tau<:X
  &\;\leadsto\;\{\tau<:h(\tau)\}
    \quad[X\mapsto h(\tau)].
\end{aligned}
\]
\end{minipage}
