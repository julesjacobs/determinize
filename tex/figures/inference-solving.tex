\smallskip
\textbf{(a) Subtype constraint reduction}
\[
\begin{aligned}
\tyR a<:\tyR b
  &\;\leadsto\;\{a\le b\}\\[2pt]
(\tau_1\to\tau_2)<:(\sigma_1\to\sigma_2)
  &\;\leadsto\;\{\sigma_1<:\tau_1,\;\tau_2<:\sigma_2\}\\[2pt]
(\tau_1\star\tau_2)<:(\sigma_1\star\sigma_2)
  &\;\leadsto\;\{\tau_1<:\sigma_1,\;\tau_2<:\sigma_2\}
  \quad(\star\in\{\times,+\})\\[2pt]
\tyL\tau<:\tyL\sigma
  &\;\leadsto\;\{\tau<:\sigma\}\\[2pt]
\chi<:\chi
  &\;\leadsto\;\varnothing
  \quad(\chi\in\{\tyU,\tyB\}).
\end{aligned}
\]

\smallskip
\textbf{(b) Type variable resolution}

\begin{minipage}[t]{0.36\linewidth}
\vspace{0pt}
\[
\begin{aligned}
h(\tyR a)&=\tyR b,\\
h(\tau\to\sigma)&=U\to V,\\
h(\tyL\tau)&=\tyL U,\\
h(\tau\star\sigma)&=U\star V,\\
h(\chi)&=\chi.
\end{aligned}
\]
\end{minipage}\hfill
\begin{minipage}[t]{0.62\linewidth}
\vspace{0pt}
\[
\begin{aligned}
X<:X
  &\;\leadsto\;\varnothing,\\[3pt]
X<:Y\ (X\ne Y)
  &\;\leadsto\;\text{defer }X<:Y,\\[3pt]
X<:\tau
  &\;\leadsto\;\{\tau'<:\tau\}
    \quad[X\mapsto\tau'],\\[3pt]
\tau<:X
  &\;\leadsto\;\{\tau<:\tau'\}
    \quad[X\mapsto\tau'].
\end{aligned}
\]
\end{minipage}

\smallskip
\noindent
{\footnotesize
Here $\tau'=h(\tau)$ for constructor-headed $\tau$;
$b,U,V$ are fresh, $\star\in\{\times,+\}$, and
$\chi\in\{\tyU,\tyB\}$. Apply substitutions throughout
and revisit deferred constraints.
}

\medskip
\textbf{Mode constraint solving}\leanref{affinity-solving}
\[
\inferrule{a\le b\\a=\E}{b:=\E}
\qquad\qquad
\inferrule{a\le b\\b=\G}{a:=\G}
\]
\noindent
{\footnotesize
Propagate until no further assignments are forced.
Reject conflicts; otherwise, assign $\E$ to remaining
mode variables and $\tyU$ to remaining type variables.
}
