\small
\textbf{$\G$ traces $\tau\in\Trace$}\leanref{traces}
\[
 \tau ::= []\mid(D,r)::\tau,
 \qquad
 p_{D,r}(\tau,s)=((D,r)::\tau,s).
\]
\textbf{Refined output measures}\leanref{trace-semantics}
\[
\begin{aligned}
 \joint e0
 &={}
 \begin{cases}
   \delta_{([],r)} & e=r,\ r\in\Real,\\
   0 & \text{otherwise},
 \end{cases}
 &\qquad
 \joint e{n+1}
 &={}
 \begin{cases}
   \joint{C[e_2]}n
     & e=C[e_1],\ e_1\rightsquigarrow e_2,\\[8pt]
   \displaystyle\int_{\Real}\joint{C[r]}n\,D(\bar v)(dr)
     & e=C[\draw D\E{\bar v}],\ \operatorname{Domain}_D(\bar v),\\[8pt]
   \displaystyle\int_{\Real}(p_{D,r})_*\joint{C[r]}n\,D(\bar v)(dr)
     & e=C[\draw D\G{\bar v}],\ \operatorname{Domain}_D(\bar v),\\[8pt]
   0 & \text{otherwise}.
 \end{cases}
\end{aligned}
\]
\[
 J_e=\sum_{n=0}^{\infty}\joint e n,
 \qquad
 \trlaw e(A)=J_e(A\times\Real)\leanref{trace-law},
 \qquad
 \law e(B)=J_e(\Trace\times B)\leanref{trace-erasure}.
\]
\caption{Trace refinement of \cref{fig:semantics}.
$\joint e n$ records the trace and output of executions that successfully return after exactly
$n$ steps. An empty trace is returned at a real value, an $\E$-sample is integrated out without changing the trace, while $(p_{D,r})_*$ prepends $(D,r)$ to each trace.
Sampling requires $\operatorname{Domain}_D(\bar v)$. Summing over $n$ gives $J_e$, whose trace and output
marginals are $\trlaw e$ and $\law e$, respectively.}
\label{fig:traces}
\Description{Traces record primitive names and values of G draws. The refined
semantics returns an empty trace at a real value, averages over E samples without
recording them, and prepends each G sample to the continuation trace. Summing over
termination depths gives the joint law; its marginals give traces and outputs.}
\end{figure}
