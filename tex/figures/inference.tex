\begin{mathpar}
\inferrule*[right=Real]
 {r\in\Real}
 {\inferconstraints{r}{\tyR a}{\varnothing}}
\and
\inferrule*[right=Add]
 {\inferconstraints{e_1}{\tau_1}{C_1}\\
  \inferconstraints{e_2}{\tau_2}{C_2}}
 {\inferconstraints{e_1+e_2}{\tyR a}{C_1\cup C_2\cup\{\tau_1<:\tyR a,\ \tau_2<:\tyR a\}}}
\and
\inferrule*[right=Mul]
 {\inferconstraints{e_1}{\tau_1}{C_1}\\
  \inferconstraints{e_2}{\tau_2}{C_2}}
 {\inferconstraints{e_1\cdot e_2}{\tyR a}{C_1\cup C_2\cup\{\tau_1<:\tyR\G,\ \tau_2<:\tyR a\}}}
\and
\inferrule*[right=Cmp]
 {\inferconstraints{e_1}{\tau_1}{C_1}\\
  \inferconstraints{e_2}{\tau_2}{C_2}}
 {\inferconstraints{e_1<e_2}{\tyB}{C_1\cup C_2\cup\{\tau_1<:\tyR\G,\ \tau_2<:\tyR\G\}}}
\and
\inferrule*[right=Uniform]
 {\inferconstraints{e_1}{\tau_1}{C_1}\\
  \inferconstraints{e_2}{\tau_2}{C_2}}
 {\inferconstraints{\kw{uniform}(e_1,e_2)}{\tyR a}{C_1\cup C_2\cup\{\tau_1<:\tyR a,\ \tau_2<:\tyR a\}}}
\and
\inferrule*[right=Gaussian]
 {\inferconstraints{e_1}{\tau_1}{C_1}\\
  \inferconstraints{e_2}{\tau_2}{C_2}}
 {\inferconstraints{\kw{gaussian}(e_1,e_2)}{\tyR a}{C_1\cup C_2\cup\{\tau_1<:\tyR a,\ \tau_2<:\tyR\G\}}}
\and
\inferrule*[right=Var]
 {x:\tau\in\Gamma}
 {\inferconstraints{x}{\tau}{\varnothing}}
\and
\inferrule*[right=Let]
 {\inferconstraints{e_1}{\tau_1}{C_1}\\
  \Gamma,x:\tau_1\vdash e_2\Rightarrow\tau_2\mid C_2}
 {\inferconstraints{\letin x{e_1}{e_2}}{\tau_2}{C_1\cup C_2}}
\and
\inferrule*[right=If]
 {\inferconstraints{e_0}{\tau_0}{C_0}\\
  \inferconstraints{e_1}{\tau_1}{C_1}\\
  \inferconstraints{e_2}{\tau_2}{C_2}}
 {\inferconstraints{\ite{e_0}{e_1}{e_2}}{X}{C_0\cup C_1\cup C_2\cup\{\tau_0<:\tyB,\ \tau_1<:X,\ \tau_2<:X\}}}
\end{mathpar}
\caption{Selected constraint generation rules. The judgment $\inferconstraints e\tau C$ infers type $\tau$ and generates
constraints $C$. Each affinity $a$ and type variable $X$ introduced in
an individual rule is fresh. At a sampling site, $a$ supplies the inferred
annotation; an explicit annotation $m\in\{\G,\E\}$ additionally generates
$a=m$. The full rules are given in
\cref{sec:appendix-constraint-generation}.}
\label{fig:inference}
\Description{Constraint generation for arithmetic, three representative sampling
primitives, variables, let bindings, and conditionals. Each operand's inferred type must be
a subtype of its required type. Variable lookup and let binding track types through the environment. Sampling sites share their inferred real affinity
with their annotation; explicit annotations fix that affinity. Conditionals require a Boolean guard and a common supertype of the branch results.}
