\textbf{Dungeon crawl: source (annotations inferred)}
\begin{lstlisting}
let crawl = fun loot =>
  (rec visit depth =>
    if flip[G](0.25) then 0
    else loot depth + visit (depth + 1)
  ) 1
in
crawl (fun depth => 200 * depth * bernoulli[E](0.01))
\end{lstlisting}
