\textbf{Residual expression reduction}
\[
\begin{gathered}
  (\lambda x.\widehat e)\,\widehat v
  \rightsquigarrow_{\mathrm{sym}}
  \widehat e[x\mapsto\widehat v]
  \qquad
  \letin x{\widehat v}{\widehat e}
  \rightsquigarrow_{\mathrm{sym}}
  \widehat e[x\mapsto\widehat v],
  \\[3pt]
  \avg D{\sym{\bar a}}
  \rightsquigarrow_{\mathrm{sym}}
  \sym{\operatorname{Mean}^{\#}_D(\bar a)}.
\end{gathered}
\]

\textbf{Symbolic small-step semantics}
\vspace{-5pt}

\begin{mathpar}
  \inferrule*[right=Reduce]
    {\widehat e_1\rightsquigarrow_{\mathrm{sym}}
      \widehat e_2}
    {
      (\sigma\mid\widehat C[\widehat e_1])\\
      \deterministicarrow_{\mathrm{sym}}
      (\sigma\mid\widehat C[\widehat e_2])
    }
  \and
  \inferrule*[right=Sample-E]
    {u\notin\operatorname{dom}(\sigma) \\
     \operatorname{Params}^{\#}_D(\bar a)}
    {
      (\sigma\mid\widehat C[
        \draw D\E{\sym{\bar a}}])\\
      \deterministicarrow_{\mathrm{sym}}
      (\sigma,u\sim D(\bar a)\mid
        \widehat C[\sym u])
    }
  \and
  \inferrule*[right=Sample-G]
    {\operatorname{Domain}_D(\bar r)}
    {
      (\sigma\mid\widehat C[
        \draw D\G{\sym{\bar r}}])\\
      \samplingarrow{D(\bar r)}_{\mathrm{sym}}
      \bigl(r\mapsto
      (\sigma\mid\widehat C[\sym r])\bigr)
    }
\end{mathpar}
\vspace{-7pt}

\caption{Symbolic semantics. Residual expression reduction
is shown above, and symbolic small-step semantics below.
\textsc{Reduce} and \textsc{Sample-E} specify deterministic
successors, while \textsc{Sample-G} specifies a sampling
action with a continuation mapping sampled values to
successor states. Symbolic evaluation contexts $\widehat C$
have the same grammar as the evaluation contexts $C$ in
\cref{fig:semantics}.}
\label{fig:symbolic-semantics}
